local function setProperty(obj, key, value)
    if not obj then
        return false
    end

    local newIndex = getMetaMethod(obj, "__newindex")
    if newIndex then
        local ok = pcall(newIndex, obj, key, value)
        if ok then
            return true
        end
    end

    if sethiddenproperty then
        local ok = pcall(sethiddenproperty, obj, key, value)
        if ok then
            return true
        end
    end

    local ok = pcall(function()
        obj[key] = value
    end)
    return ok
end

local function pickTarget(options)
    options = options or {}

    local originRoot = options.originRoot or getgenv().getHumanoidRootPart(player)
    if not originRoot then
        return
    end

    local radius = options.searchRadius or (getgenv().config and getgenv().config.searchRadius) or math.huge
    local checkForceField = options.checkForceField == true
    local checkTeam = options.checkTeam == true
    local usePrediction = options.usePrediction == true
    local useHealthPenalty = options.useHealthPenalty == true

    local bestPlayer
    local bestCharacter
    local bestRoot
    local bestHead
    local bestScore = math.huge
    local myPosition = originRoot.Position
    local myTeam = player.Team

    for _, targetPlayer in ipairs(players:GetPlayers()) do
        if targetPlayer ~= player and targetPlayer.Character then
            local character = targetPlayer.Character
            local rootPart = character:FindFirstChild("HumanoidRootPart")
            local head = character:FindFirstChild("Head")
            local humanoid = character:FindFirstChildOfClass("Humanoid")
            local hasForceField = character:FindFirstChildOfClass("ForceField") ~= nil
            local differentTeam = myTeam == nil or targetPlayer.Team == nil or myTeam ~= targetPlayer.Team

            if rootPart and humanoid and humanoid.Health > 0
                and (not usePrediction or head)
                and (not checkForceField or not hasForceField)
                and (not checkTeam or differentTeam) then

                local distance = (myPosition - rootPart.Position).Magnitude
                if distance <= radius then
                    local score = distance

                    if usePrediction or useHealthPenalty then
                        local targetPosition = rootPart.Position
                        if usePrediction then
                            targetPosition = targetPosition + rootPart.AssemblyLinearVelocity * 0.1
                        end

                        local headDistance = head and (myPosition - head.Position).Magnitude or distance
                        local predictedDistance = (myPosition - targetPosition).Magnitude
                        score = headDistance * 0.6 + predictedDistance * 0.4

                        if useHealthPenalty then
                            score = score + math.clamp(humanoid.Health / (humanoid.MaxHealth + 1), 0, 1) * 200
                        end
                    end

                    if score < bestScore then
                        bestScore = score
                        bestPlayer = targetPlayer
                        bestCharacter = character
                        bestRoot = rootPart
                        bestHead = head
                    end
                end
            end
        end
    end

    return bestPlayer, bestCharacter, bestRoot, bestHead
end

function manipulation.isAbove(fighter)
    if not fighter then
        return true
    end
    local okObs, obs = pcall(function()
        return rawget(fighter, "itemObserver") or rawget(fighter, "ItemObserver")
    end)
    if not okObs or not obs then
        return true
    end
    local ok, result = pcall(function()
        local equipped = obs:GetEquippedItem()
        if equipped ~= nil and equipped.name == "Riot Shield" then
            local pitch = math.deg(fighter:GetCameraRotation().X)
            if 22 < pitch and pitch < 91 then
                return "Below"
            end
            return "Above"
        end
        for _, it in pairs(obs:GetItems()) do
            if it.name == "Riot Shield" then
                local pitch = math.deg(fighter:GetCameraRotation().X)
                if 315 < pitch and pitch < 360 or 0 < pitch and pitch < 91 then
                    return "Above"
                end
                return "Below"
            end
        end
        return "None"
    end)
    return not (ok and result == "Below")
end

function manipulation.newShootLock()
    local lock = { lockedUntil = nil }
    function lock.shouldFire(shouldLock, duration)
        local now = os.clock()
        local locked = lock.lockedUntil ~= nil and now < lock.lockedUntil
        if shouldLock then
            lock.lockedUntil = now + duration
        end
        return locked or shouldLock
    end
    return lock
end

function manipulation.checkFighter(fighter)
    local owner = rawget(fighter, "Player")
    local entity = rawget(fighter, "Entity")
    local model = entity and rawget(entity, "Model") or nil
    if not model then
        return nil
    end
    if owner ~= nil and not manipulation.isEnemyPlayer(owner) then
        return nil
    end
    local hitboxHead = model:FindFirstChild("HitboxHead")
    local rootPart = model:FindFirstChild("HumanoidRootPart") or model:FindFirstChild("HitboxBody")
    if not (hitboxHead and rootPart) then
        return nil
    end
    local humanoid = model:FindFirstChildOfClass("Humanoid")
    if humanoid ~= nil and humanoid.Health <= 0 then
        return nil
    end
    if manipulation.isInvincible(entity) then
        return nil
    end
    local observer = rawget(fighter, "itemObserver") or rawget(fighter, "ItemObserver")
    if manipulation.isDeflecting(observer) then
        return nil
    end
    return { fighter = fighter, model = model, hitboxHead = hitboxHead, rootPart = rootPart, itemObserver = observer }
end

function manipulation.spatialTick(target)
    local now = os.clock()
    local measurement = manipulation.spatialMeasurements[target.fighter]
    if measurement == nil then
        measurement = { ExpectedDuration = 1 }
        manipulation.spatialMeasurements[target.fighter] = measurement
    end
    local entryTime = measurement.LimitEntryTime
    if not manipulation.isOutOfBounds(target.rootPart.Position) then
        if entryTime ~= nil then
            if manipulation.targetHasAmmo(target) then
                measurement.ExpectedDuration = now - entryTime
            end
            measurement.LimitEntryTime = nil
        end
        return false
    end
    if entryTime == nil then
        measurement.LimitEntryTime = now
        entryTime = now
    end
    if manipulation.targetHasAmmo(target) then
        if measurement.ExpectedDuration - manipulation.stability <= now - entryTime then
            return false
        end
    end
    return true
end

function manipulation.hitscanShoot(lock, fireRate, repeatCount)

    end
end

function manipulation.send(item, tokenName, payload)
    if payload == nil then
        return false
    end
    local enums = requireOptionalModule(replicatedStorage.Modules.EnumLibrary)
    local remote = replicatedStorage.Remotes.Replication.Fighter.UseItem

    local token = enums and enums:ToEnum(tokenName)
    if not remote or not token or not objectId then
        return false
    end
    manipulation.nativeFire(remote, objectId, token, payload, nil)
    return true
end

local function readProperty(instance, propertyName)
    if not instance then
        return
    end

    local ok, value = pcall(function()
        return instance[propertyName]
    end)
    if ok and value ~= nil then
        return value
    end

    local index = getMetaMethod(instance, "__index")
    if index then
        ok, value = pcall(index, instance, propertyName)
        if ok and value ~= nil then
            return value
        end
    end

    if gethiddenproperty then
        ok, value = pcall(gethiddenproperty, instance, propertyName)
        if ok and value ~= nil then
            return value
        end
    end

    if ficlone then
        ok, value = pcall(function()
            local clone = ficlone(instance)
            return clone and clone[propertyName]
        end)
        if ok and value ~= nil then
            return value
        end
    end

    if getproperties then
        ok, value = pcall(function()
            local properties = getproperties(instance)
            return properties and properties[propertyName]
        end)
        if ok and value ~= nil then
            return value
        end
    end
end

function isNaNValue(v)
    return v ~= v
end
function isInfiniteValue(v)
    return math.abs(v) == math.huge
end
function finiteValue(v)
    return not isNaNValue(v) and not isInfiniteValue(v)
end
function validCoordinates(x, y, z)
    return finiteValue(x) and finiteValue(y) and finiteValue(z)
end
function validVector(v)
    return v and validCoordinates(v.X, v.Y, v.Z)
end

local gunPerfectOriginals = setmetatable({}, { __mode = "k" })

local function gunPerfectWanted()

local function applyGunPerfectAccuracy()
    local loaded, itemLibrary = pcall(function()
        return require(replicatedStorage.Modules.ItemLibrary)
    end)
    if not loaded or typeof(itemLibrary) ~= "table" then
        return
    end
    local visited = {}
    local function setPerfect(tbl)
        visited[tbl] = true
        for _, v in pairs(tbl) do
            if typeof(v) == "table" and not visited[v] then
                for _, key in ipairs(gunPerfectKeys) do
                    if v[key] then
                        local saved = gunPerfectOriginals[v]
                        if not saved then
                            saved = {}
                            gunPerfectOriginals[v] = saved
                        end
                        if saved[key] == nil then
                            saved[key] = v[key]
                        end
                        pcall(function() v[key] = 0 end)
                    end
                end
                setPerfect(v)
            end
        end
    end
    pcall(setPerfect, itemLibrary)
end
local function restoreGunPerfectAccuracy()
    for tbl, saved in pairs(gunPerfectOriginals) do
        gunPerfectOriginals[tbl] = nil
        for key, value in pairs(saved) do
            pcall(function() tbl[key] = value end)
        end
    end
end
local function reconcileGunPerfect(force)
    if force then
        restoreGunPerfectAccuracy()
        gunPerfectState.applied = false
    end
    local wanted = gunPerfectWanted()
    if wanted and not gunPerfectState.applied then
        gunPerfectState.applied = true
        applyGunPerfectAccuracy()
    elseif not wanted and gunPerfectState.applied then
        gunPerfectState.applied = false
        restoreGunPerfectAccuracy()
    end
end

function fastMelee.apply(info)
    local saved = fastMelee.saved[info]
    if not saved then
        saved = {}
        fastMelee.saved[info] = saved
    end
    for k, v in pairs(fastMelee.cfg) do
        if saved[k] == nil then
            local current = info[k]
            if current == nil then
                saved[k] = fastMelee.absent
            else
                saved[k] = current
            end
        end
        info[k] = v
    end
end
function fastMelee.restore(info)
    local saved = fastMelee.saved[info]
    if not saved then
        return
    end
    fastMelee.saved[info] = nil
    for k, v in pairs(saved) do
        if v == fastMelee.absent then
            info[k] = nil
        else
            info[k] = v
        end
    end
end
function fastMelee.restoreAll()
    for info in pairs(fastMelee.saved) do
        fastMelee.restore(info)
    end
end
function fastMelee.wanted()
    return (config.fastMelee or fastMelee.voidOwned) and true or false
end
function fastMelee.install()
    if fastMelee.hooked or fastMelee.hooking then
        return
    end
    fastMelee.hooking = true
    task.spawn(function()
        pcall(function()
            local clientItem = require(player.PlayerScripts.Modules.ClientReplicatedClasses.ClientFighter.ClientItem)
            local input = clientItem.Input
            local captured = nil
            captured = hookfunction(input, newcclosure(function(...)
                local args = { ... }
                if type(args[1]) == "table" and type(args[1].Info) == "table" then
                    if fastMelee.wanted() then
                        fastMelee.apply(args[1].Info)
                    else
                        fastMelee.restore(args[1].Info)
                    end
                end
                return captured(...)
            end))
            fastMelee.original = captured
            fastMelee.hooked = true
        end)
        fastMelee.hooking = false
        if not fastMelee.wanted() then
            fastMelee.restoreAll()
            fastMelee.remove()
        end
    end)
end
function fastMelee.remove()
    if not fastMelee.hooked or not fastMelee.original then
        return
    end
    local original = fastMelee.original
    fastMelee.original = nil
    fastMelee.hooked = false
    pcall(function()
        local clientItem = require(player.PlayerScripts.Modules.ClientReplicatedClasses.ClientFighter.ClientItem)
        hookfunction(clientItem.Input, original)
    end)
end
function fastMelee.reconcile()
    if fastMelee.wanted() then
        fastMelee.install()
    else
        fastMelee.restoreAll()
        fastMelee.remove()
    end
end

local gunMeleeBypass = { owners = {}, conn = nil, added = nil, removed = nil, projectiles = {}, targetPos = CFrame.new(9000, 9000, 9000) }
local function stopGunMeleeBypassLogic()
    gunMeleeBypass.conn = disconnect(gunMeleeBypass.conn)
    gunMeleeBypass.added = disconnect(gunMeleeBypass.added)
    gunMeleeBypass.removed = disconnect(gunMeleeBypass.removed)
    gunMeleeBypass.projectiles = {}
end
local function startGunMeleeBypassLogic()
    local bypass = gunMeleeBypass
    if bypass.conn then
        return
    end
    bypass.added = workspace.ChildAdded:Connect(function(o)
        if not o:IsA("BasePart") then
            return
        end
        if o.Name == "CoreProjectile" then
            bypass.projectiles[o] = true
        elseif o.Name == "Part" then
            task.defer(function()
                if o and o.Parent and o.AssemblyLinearVelocity.Magnitude > 50 then
                    bypass.projectiles[o] = true
                end
            end)
        end
    end)
    bypass.removed = workspace.ChildRemoved:Connect(function(o)
        bypass.projectiles[o] = nil
    end)
    for _, meleeObj in ipairs(workspace:GetChildren()) do
        if meleeObj.Name == "CoreProjectile" and meleeObj:IsA("BasePart") then
            bypass.projectiles[meleeObj] = true
        end
    end
    bypass.conn = runService.Heartbeat:Connect(function()
        if next(bypass.owners) == nil then
            stopGunMeleeBypassLogic()
            return
        end
        pcall(function()
            for _, meleePlayer in ipairs(players:GetPlayers()) do
                if meleePlayer ~= player and meleePlayer.Character then
                    local meleeRoot = meleePlayer.Character:FindFirstChild("HumanoidRootPart")
                    if meleeRoot then
                        meleeRoot.CFrame = bypass.targetPos
                        meleeRoot.AssemblyLinearVelocity = zeroVector
                    end
                end
            end
            for meleeProj in pairs(bypass.projectiles) do
                if meleeProj and meleeProj.Parent then
                    meleeProj.CFrame = bypass.targetPos
                    meleeProj.AssemblyLinearVelocity = zeroVector
                else
                    bypass.projectiles[meleeProj] = nil
                end
            end
        end)
    end)
end

    local function hitAllVoidTarget(root)
        local cfg = hitAllVoidCfg
        local x = cfg.baseX + (math.random() * 2 - 1) * cfg.radius
        local y = (math.random() * 2 - 1) * (cfg.yMax - cfg.yMin) / 2 + (cfg.yMax + cfg.yMin) / 2
        local z = cfg.baseZ + (math.random() * 2 - 1) * cfg.radius
        if x ~= x or x == math.huge or x == -math.huge then
            x = cfg.baseX
        end
        if y ~= y or y == math.huge or y == -math.huge then
            y = 0
        end
        if z ~= z or z == math.huge or z == -math.huge then
            z = cfg.baseZ
        end
        y = math.clamp(y, cfg.yMin, cfg.yMax)
        local radius = math.abs(cfg.radius)
        local dx, dz = x - cfg.baseX, z - cfg.baseZ
        local flat = math.sqrt(dx * dx + dz * dz)
        if flat > radius and flat > 0 then
            local scale = radius / flat
            x = cfg.baseX + dx * scale
            z = cfg.baseZ + dz * scale
        end
        if cfg.safeRadius > 0 and root then
            local me = root.Position
            local ddx, ddy, ddz = x - me.X, y - me.Y, z - me.Z
            local dist = math.sqrt(ddx * ddx + ddy * ddy + ddz * ddz)
            if dist < cfg.safeRadius then
                if dist < 0.01 then
                    ddx, ddy, ddz, dist = 1, 0, 0, 1
                end
                local scale = cfg.safeRadius / dist
                x = me.X + ddx * scale
                y = me.Y + ddy * scale
                z = me.Z + ddz * scale
            end
        end

function voidPattern.axisCap()
    local spread = math.clamp(tonumber(config.motionDistance) or 300, 1, 5000)
    return spread / 300 * 2000000000
end
function voidPattern.reset()
    getgenv().voidPatternState = nil
end
function voidPattern.heading(elapsed)
    local fx, fz = 0, 0
    local amp, freq = 1, 0.0001
    for _ = 1, 4 do
        fx = fx + math.noise(elapsed * freq, 0) * amp
        fz = fz + math.noise(0, elapsed * freq) * amp
        freq = freq * 2.37
        amp = amp * 0.5
    end
    fx = fx + math.sin(elapsed * 0.00213) * math.cos(elapsed * 0.00344) * 0.2
    fz = fz + math.cos(elapsed * 0.00131) * math.sin(elapsed * 0.00579) * 0.2
    local mag = math.sqrt(fx * fx + fz * fz)
    if mag < 0.001 then
        return math.cos(elapsed * 0.1), math.sin(elapsed * 0.1)
    end
    return fx / mag, fz / mag
end
function voidPattern.state(cap)
    local env = getgenv()
    local st = env.voidPatternState
    if not st then
        local rnd = math.random
        st = {
            at = nil,
            t = rnd() * 1000,
            clock = 0,
            gX = (rnd() * 2 - 1) * cap * 0.05,
            gZ = (rnd() * 2 - 1) * cap * 0.05,
            yBase = (rnd() * 2 - 1) * cap * 0.3,
            gYOff = 0,
            yDir = rnd() < 0.5 and 1 or -1,
            dirX = rnd() * 2 - 1,
            dirZ = rnd() * 2 - 1,
        }
        env.voidPatternState = st
    end
    return st
end
function voidPattern.offset(preset, t, cap)
    local kind = preset.kind
    if kind == "circle" then
        local a = t * preset.rate
        local r = cap * preset.radius
        return math.cos(a) * r, math.sin(a * 2) * r * preset.yAmp, math.sin(a) * r
    elseif kind == "eight" then
        local a = t * preset.rate
        local r = cap * preset.radius
        return math.sin(a) * r, math.sin(a * 2) * r * preset.yAmp, math.sin(a * 2) * r
    elseif kind == "grid" then
        local index = math.floor(t * preset.hz)
        local half = math.floor(preset.size / 2)
        local cell = cap * preset.cell
        local col = index % preset.size - half
        local row = math.floor(index / preset.size) % preset.size - half
        return col * cell, (index % 2 == 0) and 0 or cell * preset.yStep, row * cell
    elseif kind == "steps" then
        local index = math.floor(t * preset.hz)
        local half = math.floor(preset.width / 2)
        local cell = cap * preset.cell
        local col = index % preset.width - half
        local row = math.floor(index / preset.width) % preset.width - half
        return col * cell, (index % preset.layers) * cell * preset.yStep, row * cell
    end
    return 0, 0, 0
end
function voidPattern.step(st, presetName, cap)
    local now = os.clock()
    local dt = st.at and math.min(now - st.at, 0.25) or 0
    st.at = now
    st.t = st.t + dt * 60
    st.clock = st.clock + dt
    local base = voidPattern.presets["Smooth Drift"]
    local headX, headZ = voidPattern.heading(st.t)
    local blend = math.min(base.blend * dt * 10, 1)
    local norm = cap * base.pull
    st.dirX = st.dirX + (headX - st.gX / norm - st.dirX) * blend
    st.dirZ = st.dirZ + (headZ - st.gZ / norm - st.dirZ) * blend
    st.gX = st.gX + st.dirX * cap * base.speed * dt
    st.gZ = st.gZ + st.dirZ * cap * base.speed * dt
    st.gYOff = st.gYOff + st.yDir * cap * base.yRate * dt
    if cap * base.yRange <= math.abs(st.gYOff) then
        st.yDir = -st.yDir
    end
    local preset = voidPattern.presets[presetName] or voidPattern.presets["Mixed"]
    if preset.kind == "mixed" then
        local sequence = preset.sequence
        preset = voidPattern.presets[sequence[math.floor(st.clock / preset.period) % #sequence + 1]]
    end
    local ox, oy, oz = voidPattern.offset(preset, st.clock, cap)
    local limit = cap * 0.95
    local ax = math.clamp(st.gX + ox, -limit, limit)
    local ay = math.clamp(st.yBase + st.gYOff + oy, -limit, limit)
    local az = math.clamp(st.gZ + oz, -limit, limit)
    return ax, ay, az
end

local voidSpinCo = nil
getgenv().teleportationIsValid = function(p, teamCheck, keepFf)
    if not p or p == player then
        return false
    end
    local ch = p.Character
    if not ch then
        return false
    end
    local targetRootPart = ch:FindFirstChild("HumanoidRootPart")
    local targetHumanoid = ch:FindFirstChildOfClass("Humanoid")
    if not targetRootPart or not targetHumanoid or targetHumanoid.Health <= 0 then
        return false
    end
    if not keepFf and ch:FindFirstChildOfClass("ForceField") then
        return false
    end
    if teamCheck and player.Team ~= nil and p.Team ~= nil and player.Team == p.Team then
        return false
    end
    return true, targetRootPart, targetHumanoid
end
getgenv().teleportationScore = function(targetRootPart, targetHumanoid, playerPosition, mode)
    local distanceValue = (targetRootPart.Position - playerPosition).Magnitude
    if mode == "Lowest Health" then
        return targetHumanoid.Health + distanceValue * 0.000001
    elseif mode == "Closest To Cursor" then
        local cam = workspace.CurrentCamera
        if cam then
            local v, on = cam:WorldToViewportPoint(targetRootPart.Position)
            local mouseLocation = game:GetService("UserInputService"):GetMouseLocation()
            local screenDistance = (Vector2.new(v.X, v.Y) - mouseLocation).Magnitude
            if not on then
                screenDistance = screenDistance + 100000
            end
            return screenDistance + distanceValue * 0.000001
        end
    end
    return distanceValue
end

getgenv().startAutoCollect = function()
    autoCollectConnection = disconnect(autoCollectConnection)
    local autoCollectCachedParts = {}
    local autoCollectCacheLastUpdate = 0
    local autoCollectScanned = false
    local function autoCollectConsider(o)
        if o:IsA("BasePart") then
            local n = string.lower(o.Name)
            local itemVal = 0
            for _, Kw in ipairs(getgenv().autoCollectKeywordList) do
                if n:Find(Kw, 1, true) then
                    local v = getgenv().autoCollectKeywordValues[Kw]
                    if v > itemVal then
                        itemVal = v
                    end
                end
            end
            if itemVal > 0 then
                autoCollectCachedParts[#autoCollectCachedParts + 1] = { partRef = o, secondaryValue = itemVal }
            end
        end
    end
    getgenv().autoCollectAddedConn = disconnect(getgenv().autoCollectAddedConn)
    getgenv().autoCollectAddedConn = workspace.DescendantAdded:Connect(function(o)
        if autoCollectScanned then
            pcall(autoCollectConsider, o)
        end
    end)
    autoCollectConnection = runService.Heartbeat:Connect(function()
        if not config.autoCollect then
            return
        end
        local playerRootPart = getgenv().getHumanoidRootPart(player)
        if not playerRootPart then
            return
        end
        local currentTime = tick()
        if currentTime - getgenv().autoCollectLastTick < 0.5 then
            return
        end
        getgenv().autoCollectLastTick = currentTime
        if not autoCollectScanned then
            autoCollectScanned = true
            autoCollectCacheLastUpdate = currentTime
            for _, o in ipairs(workspace:GetDescendants()) do
                autoCollectConsider(o)
            end
        elseif currentTime - autoCollectCacheLastUpdate >= 5 then
            autoCollectCacheLastUpdate = currentTime
            local alive = {}
            for _, entry in ipairs(autoCollectCachedParts) do
                if entry.partRef and entry.partRef.Parent then
                    alive[#alive + 1] = entry
                end
            end
            autoCollectCachedParts = alive
        end
        local best, bestScore = nil, math.huge
        local playerPosition = playerRootPart.Position
        local cR = config.collectRadius or 120
        for _, entry in ipairs(autoCollectCachedParts) do
            local o = entry.partRef
            if o and o.Parent then
                local callSucceeded, opos = pcall(function() return o.Position end)
                if callSucceeded and opos then
                    local distanceValue = (playerPosition - opos).Magnitude
                    if distanceValue < cR then
                        local score = distanceValue / entry.secondaryValue
                        if score < bestScore then
                            bestScore = score
                            best = o
                        end
                    end
                end
            end
        end
        if best then
            setCFrame(playerRootPart, CFrame.new(best.Position), Vector3.zero)
        end
    end)
end

local function antiAimAdvanceYaw()
    local now = os.clock()
    local dt = 1 / 60
    if antiAimLastStep > 0 then
        dt = math.clamp(now - antiAimLastStep, 0, 0.1)
    end
    antiAimLastStep = now
    local speed = tonumber(config.antiAimSpeed) or 30
    if config.riotAbuser == true then
        speed = tonumber(config.riotAbuserSpinSpeed) or speed
    end
    antiAimYaw = (antiAimYaw + math.rad(speed) * dt * 60) % (math.pi * 2)
    return antiAimYaw
end
local function antiAimReleaseHumanoid()
    local characterHumanoid = antiAimSavedHumanoid
    local saved = antiAimSavedAutoRotate
    antiAimSavedHumanoid = nil
    antiAimSavedAutoRotate = nil
    if characterHumanoid and characterHumanoid.Parent and saved ~= nil then
        pcall(function() characterHumanoid.AutoRotate = saved end)
    end
end
local function antiAimIsBodyPart(inst, char)
    local parent = inst.Parent
    while parent and parent ~= char do
        if parent.Name == "EquippedItem" or parent:IsA("Tool") then
            return false
        end
        parent = parent.Parent
    end
    return true
end
local function antiAimReassert()
    local list = antiAimPartList

    if not sendCFrame or #list == 0 then
        return

    for i = 1, #list do
        local part = list[i]
        if part and part.Parent then
            setProperty(part, "CFrame", sendCFrame)
        end
    end
end

function startAntiAimHooks()
    if getgenv().antiAimHooksActive then
        return
    end

    local modules = replicatedStorage:FindFirstChild("Modules")

    pcall(function()

        utility = require(modules:FindFirstChild("Utility", true))
        cameraCtrl = require(scripts:FindFirstChild("CameraController", true))
    end)
    pcall(function()
        antiAimRotationFunction = mechCtrl and mechCtrl._UpdateCharacterRotation
        if not antiAimRotationFunction then
            return
        end
        antiAimRotOriginal = hookfunction(antiAimRotationFunction, function(self, ...)
            local riotFaceApply = getgenv().riotFaceApply
            if riotFaceApply then
                riotFaceApply()
            end
            local perfectBlockApply = getgenv().perfectBlockApply
            if perfectBlockApply then
                perfectBlockApply()
            end
            if (antiAimOn() and config.antiAimSpin ~= false) or config.riotAbuser == true then
                if self.IsSliding or not self:IsAlive() then
                    return
                end
                local fightObject = self.LocalFighter
                local entity = fightObject and fightObject.Entity
                if not entity then
                    return
                end
                local humanoid = entity.Humanoid
                local rootPart = entity.RootPart
                if not humanoid or not rootPart then
                    return
                end
                local publicState = cameraCtrl and cameraCtrl:GetPublicState()
                local cameraState = cameraCtrl and cameraCtrl.CameraState
                local states = cameraState and cameraState.States
                local freeCamState = states and states.CustomFreecam
                local thirdPersonUnlockedState = states and states.ThirdPersonUnlockedMouse
                if (not entity:IsEmoting()) or (publicState and (publicState ~= freeCamState and publicState ~= thirdPersonUnlockedState)) then
                    if antiAimSavedHumanoid ~= humanoid then
                        antiAimReleaseHumanoid()
                        antiAimSavedHumanoid = humanoid
                        antiAimSavedAutoRotate = humanoid.AutoRotate
                    end
                    humanoid.AutoRotate = false
                    local yaw = antiAimAdvanceYaw()
                    local collapsedRoot = antiAimRootIndex > 0 and antiAimPartList[antiAimRootIndex] == rootPart
                    if collapsedRoot then
                        antiAimPartCFrames[antiAimRootIndex] = CFrame.new(antiAimPartCFrames[antiAimRootIndex].Position) * CFrame.Angles(0, yaw, 0)
                    else
                        local rootPos = rootPart.Position
                        if rootPos.X == rootPos.X and rootPos.Y == rootPos.Y and rootPos.Z == rootPos.Z then
                            rootPart.CFrame = CFrame.new(rootPos) * CFrame.Angles(0, yaw, 0)
                            rootPart.AssemblyAngularVelocity = Vector3.zero
                        end
                    end
                end
                return
            end
            if antiAimSavedHumanoid then
                antiAimReleaseHumanoid()
            end
            return antiAimRotOriginal(self, ...)
        end)
    end)
    pcall(function()
        antiAimEncodeFunction = utility and utility.EncodeCameraRotation
        if not antiAimEncodeFunction then
            return
        end
        antiAimEncodeOriginal = hookfunction(antiAimEncodeFunction, function(self, rotation, ...)
            if antiAimOn() then
                if config.antiAimArmspin then
                    antiAimArmAngle = (antiAimArmAngle + ((tonumber(config.antiAimSpeed) or 30) * 0.005)) % math.pi
                end
                rotation = Vector2.new(antiAimPitchValue(), antiAimArmAngle + (tonumber(config.antiAimYawOffset) or 0))
            end
            return antiAimEncodeOriginal(self, rotation, ...)
        end)
    end)
    pcall(function()
        local origV2new = getrenv().Vector2.new
        antiAimVec2Original = hookfunction(origV2new, function(x, y, ...)
            if antiAimOn() then
                local caller, source = debug.info(3, "ns")
                if caller == "Update" and string.find(tostring(source or ""), ".Joints") then
                    local stackSelf = getstack(3, 1)
                    if stackSelf then
                        local cliFightCl = stackSelf.ClientFighterCharacter
                        if cliFightCl and cliFightCl.Player == player then
                            x = antiAimPitchValue()
                            local yawOffset = tonumber(config.antiAimYawOffset) or 0
                            if yawOffset ~= 0 and type(y) == "number" then
                                y = y + yawOffset
                            end
                        end
                    end
                end
            end
            return antiAimVec2Original(x, y, ...)
        end)
    end)
    antiAimCollapseConnection = disconnect(antiAimCollapseConnection)
    antiAimCollapseConnection = runService.Heartbeat:Connect(function()
        if #antiAimPartList > 0 then
            antiAimRestore()
        end
        if config.antiAim ~= true then
            return
        end
        local char = player.Character
        if not char then
            return
        end
        local root = char:FindFirstChild("HumanoidRootPart")
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if not root or not humanoid or humanoid.Health <= 0 then
            return
        end
        local targetLegPart = char:FindFirstChild("RightFoot")
        if not targetLegPart then
            return
        end
        local targetPos = targetLegPart.Position
        local nowClock = tick()
        local partCache = getgenv().antiAimPartCache
        if not partCache or partCache.char ~= char or nowClock - partCache.at > 0.5 then
            local parts = {}
            for _, inst in ipairs(char:GetDescendants()) do
                if inst:IsA("BasePart") and antiAimIsBodyPart(inst, char) then
                    parts[#parts + 1] = inst
                end
            end
            partCache = { char = char, at = nowClock, parts = parts }
            getgenv().antiAimPartCache = partCache
        end
        local list = {}
        local saved = {}
        local total = 0
        local rootIndex = 0
        for _, inst in ipairs(partCache.parts) do
            if inst.Parent then
                local realCFrame = readProperty(inst, "CFrame")
                if realCFrame then
                    total = total + 1
                    list[total] = inst
                    saved[total] = realCFrame
                    if inst == root then
                        rootIndex = total
                    end
                end
            end
        end
        if total == 0 then
            return
        end
        antiAimPartList = list
        antiAimPartCFrames = saved
        antiAimRootIndex = rootIndex
        local sendCFrame = CFrame.new(targetPos.X, targetPos.Y, targetPos.Z)
        antiAimSendCFrame = sendCFrame
        for i = 1, total do
            setProperty(list[i], "CFrame", sendCFrame)
        end
    end)
    pcall(function() runService:UnbindFromRenderStep(antiAimRenderName) end)
    pcall(function() runService:UnbindFromRenderStep(antiAimRenderName .. "Restore") end)
    pcall(function()
        runService:BindToRenderStep(antiAimRenderName, Enum.RenderPriority.First.Value, antiAimReassert)
    end)
    pcall(function()
        runService:BindToRenderStep(antiAimRenderName .. "Restore", Enum.RenderPriority.Last.Value, antiAimRestore)
    end)
end

function findRotationCameraTable(cached)
    if type(cached) == "table" and typeof(cached.Rotation) == "Vector2" then
        return cached
    end
    local candidates = {}
    if antiAimRotOriginal then
        candidates[#candidates + 1] = antiAimRotOriginal
    end
    if antiAimRotationFunction then
        candidates[#candidates + 1] = antiAimRotationFunction
    end
    if filtergc then
        pcall(function()
            local found = filtergc("function", { Name = "_UpdateCharacterRotation" }, true)
            if found then
                candidates[#candidates + 1] = found
            end
        end)
    end
    for _, candidate in ipairs(candidates) do
        local okUpvalues, upvalues = pcall(debug.getupvalues, candidate)
        if okUpvalues and type(upvalues) == "table" then
            for _, v in ipairs(upvalues) do
                if type(v) == "table" and typeof(v.Rotation) == "Vector2" then
                    return v
                end
            end
        end
    end
    return nil
end

            local ny = spoofY and (realPosition.Y + oy + offsetY) or realPosition.Y
            local nz = spoofZ and (realPosition.Z + oz + offsetZ) or realPosition.Z
            spoofCFrame = CFrame.new(nx, ny, nz)

            local nz = spoofZ and (realPosition.Z + rz + offsetZ) or realPosition.Z
            spoofCFrame = CFrame.new(nx, ny, nz)
        elseif mode == "Void" then
            local vx = spoofX and (math.random(-150, 150) * 1000 + offsetX) or realPosition.X
            local vy = spoofY and (math.random(10, 100) * 100 + offsetY) or realPosition.Y
            local vz = spoofZ and (math.random(-150, 150) * 1000 + offsetZ) or realPosition.Z
            spoofCFrame = CFrame.new(vx, vy, vz)
        else
            local nx = spoofX and (realPosition.X + offsetX) or realPosition.X
            local ny = spoofY and (realPosition.Y + offsetY) or realPosition.Y
            local nz = spoofZ and (realPosition.Z + offsetZ) or realPosition.Z
            spoofCFrame = CFrame.new(nx, ny, nz)
        end
        if not spoofCFrame then
            return
        end
                setCFrame(playerRootPart, spoofCFrame, true)
    end)

            local fpy = axis == "Y" or axis == "All"

local evasionRand = Random.new()
function evasionRingPoint(anchor, minRad, maxRad)
    local angle = evasionRand:NextNumber(0, 2 * math.pi)
    local radiusValue = evasionRand:NextNumber(minRad, maxRad)
    return CFrame.new(anchor + Vector3.new(math.cos(angle) * radiusValue, 0, math.sin(angle) * radiusValue))
 * CFrame.fromOrientation(
            evasionRand:NextNumber(0, 2 * math.pi),
            evasionRand:NextNumber(0, 2 * math.pi),
            evasionRand:NextNumber(0, 2 * math.pi)
        )
end
function evasionScatterFar(position)
    local baseRad = config.evasionRadiusMin or 100
    local maxRad = config.evasionRadiusMax or 150
    local n = baseRad + baseRad * ((maxRad / baseRad) - 1)
    local ring = evasionRingPoint(position, baseRad, n)
    local ringPosition = ring.Position
    local x2, y2, z = ringPosition.X, ringPosition.Y, ringPosition.Z
    local pick = evasionRand:NextInteger(1, 3)
    if pick == 1 then x2 = 1073741824
    elseif pick == 2 then y2 = 1073741824
    else z = 1073741824 end
    return ring - ringPosition + Vector3.new(x2, y2, z)
end
function getServerCFrame(playerRootPart)
    local clientCFrame = playerRootPart.CFrame
    local currentCFrame = evasionScatterFar(clientCFrame.Position)
    return CFrame.new(currentCFrame.Position)
end

local evasionCacheEnu = nil
function encodeEvasionValue(name)
    if not evasionCacheEnu then
        local mod = findEvasionChild(game:GetService("ReplicatedStorage"), "Modules", "EnumLibrary")
        if mod then
            local callSucceeded, lib = pcall(require, mod)
            if callSucceeded and type(lib) == "table" then
                evasionCacheEnu = lib
            end
        end
    end
    if not evasionCacheEnu then
        return
    end
    local callSucceeded, token = pcall(evasionCacheEnu.ToEnum, evasionCacheEnu, name)
    return callSucceeded and token or nil
end
function encodeAngleByte(n)
    if n == n then
        return utf8.char(math.clamp(math.floor(n % (2 * math.pi) / math.pi / 2 * 256 + 0.5), 0, 255))
    end
    return utf8.char(0)
end
function encodeEvasionAngles(pitch, yaw)
    return encodeAngleByte(math.rad(pitch)) .. encodeAngleByte(math.rad(yaw))
end

getgenv().startUndergroundBypass = function()
    if getgenv().internalState.undergroundBypassActive then
        return
    end
    getgenv().internalState.undergroundBypassActive = true
    config.undergroundBypass = true
    local upperBodyLast = 0
    getgenv().internalState.undergroundBypassConnection = runService.Heartbeat:Connect(function()
        if not config.undergroundBypass then
            return
        end
        local currentTime = tick()
        if currentTime - upperBodyLast < (1 / 256) then
            return
        end
        upperBodyLast = currentTime
        local playerRootPart = getgenv().getHumanoidRootPart(player)
        if not playerRootPart then
            return
        end
        local myHumanoid = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
        if not myHumanoid or myHumanoid.Health <= 0 then
            return
        end
        local function undergroundBypassRange() return 20 end
        local function undergroundBypassFindTarget(range)
                        return select(1, pickTarget({ checkForceField = false, searchRadius = range }))
        end
        local target = undergroundBypassFindTarget(undergroundBypassRange())
        if not target or not target.Character then
            return
        end
        local targetRootPart = target.Character:FindFirstChild("HumanoidRootPart")
        local head = target.Character:FindFirstChild("Head")
        if not targetRootPart then
            return
        end
        local firePart = head or targetRootPart
        local targetCharacterModel = target.Character
        local position = {}
        local rawCFramePos = targetRootPart.CFrame
        if validVector(rawCFramePos.Position) then
            position[#position + 1] = rawCFramePos.Position
        end
        local rawVelocity = sampleVoidPart(targetRootPart)
        if rawVelocity and validVector(rawVelocity) then
            position[#position + 1] = rawVelocity
        end
        if head then
            local rawHeadPosition = sampleVoidPart(head)
            if rawHeadPosition and validVector(rawHeadPosition) then
                position[#position + 1] = rawHeadPosition
            end
        end
        if #position == 0 then
            return
        end
        setFallenHeight()
            local snaps = math.max(config.undergroundBypassSnapReps or 3, 1)
            local yOff = config.undergroundBypassSnapY or 5
            for _, pos in ipairs(position) do
                if validVector(pos) then
                    local snapCFrame = CFrame.new(pos.X, pos.Y + yOff, pos.Z)
                    for _ = 1, snaps do
                        setCFrame(playerRootPart, snapCFrame, zeroVector, zeroVector)
                    end
                    myHumanoid:ChangeState(Enum.HumanoidStateType.Physics)
                myHumanoid:ChangeState(Enum.HumanoidStateType.Physics)
                manipulation.fireToward(firePart, targetCharacterModel)
            end
        end
        setFallenHeight()
    end)

function lcg(s)
    s = math.floor(math.abs(s)) % 2147483647
    if s == 0 then
        s = 1
    end
    return (s * 48271) % 2147483647
end
function lcgFloat(s)
    return (math.floor(math.abs(s)) % 2147483647) / 2147483647
end
function hashSeedString(str, salt)
    str = tostring(str or "") .. "|" .. tostring(salt or "")
    local h = 1779033703
    local g = 1013904223
    for i = 1, #str do
        local b = string.byte(str, i)
        h = (h * 48271 + b * 7919 + i) % 2147483647
        g = (g * 69621 + b * 104729 + h) % 2147483647
    end
    local x = (h + g * 31 + #str) % 2147483647
    if x == 0 then
        x = 1
    end
    for _ = 1, 8 do
        x = (x * 48271) % 2147483647
    end
    return x
end
function lcgSequence(s)
    local s1 = lcg(s)
    local s2 = lcg(s1)
    local s3 = lcg(s2)
    local s4 = lcg(s3)
    return s4, lcgFloat(s1), lcgFloat(s2), lcgFloat(s3), lcgFloat(s4)
end
local transThreshold = 110000
getgenv().localMinStep = 80000
function scoreRouteCandidate(cx, cy, cz, prevX, prevY, prevZ, route, routeLen, predX, predY, predZ)
    local score = 0
    local dx = cx - prevX
    local dy = cy - prevY
    local dz = cz - prevZ
    local stepDist = math.sqrt(dx * dx + dy * dy + dz * dz)
    if stepDist < getgenv().localMinStep then
        return -1e9
    end
    if predX then
        local px = cx - predX
        local py = cy - predY
        local pz = cz - predZ
        local predDist = math.sqrt(px * px + py * py + pz * pz)
        local predSafe = 55000000 * 5.5
        if predDist < predSafe then
            score = score - (predSafe - predDist) * 3.0
        else
            score = score + math.min((predDist - predSafe) / predSafe, 1) * 60
        end
    end
    local mag = math.sqrt(cx * cx + cy * cy + cz * cz)
    if mag < transThreshold then
        score = score - ((transThreshold - mag) / transThreshold) * 200
    else
        score = score + math.min((mag - transThreshold) / transThreshold, 1) * 40
    end
    local absY = math.abs(cy)
    if absY > 50000 then
        score = score + math.min(absY / 500000, 1) * 60
    end
    local stepNorm = stepDist / math.max(mag, 1)
    score = score + math.min(stepNorm, 1) * 50
    if routeLen > 0 then
        local prev2X = route[routeLen][1]
        local prev2Y = route[routeLen][2]
        local prev2Z = route[routeLen][3]
        local v1x = prevX - prev2X
        local v1y = prevY - prev2Y
        local v1z = prevZ - prev2Z
        local v1m = math.sqrt(v1x * v1x + v1y * v1y + v1z * v1z)
        local v2x = cx - prevX
        local v2y = cy - prevY
        local v2z = cz - prevZ
        local v2m = math.sqrt(v2x * v2x + v2y * v2y + v2z * v2z)
        if v1m > 0 and v2m > 0 then
            local dot = (v1x * v2x + v1y * v2y + v1z * v2z) / (v1m * v2m)
            score = score + (1 - math.abs(dot)) * 80
        end
    end
    local minSep = math.huge
    local checkN = math.min(routeLen, 12)
    local checkStart = routeLen - checkN + 1
    for i = checkStart, routeLen do
        if route[i] then
            local ex = cx - route[i][1]
            local ey = cy - route[i][2]
            local ez = cz - route[i][3]
            local sep = math.sqrt(ex * ex + ey * ey + ez * ez)
            if sep < minSep then
                minSep = sep
            end
        end
    end
    if minSep ~= math.huge then
        score = score + math.min(minSep / transThreshold, 2) * 70
    end
    local pmag = math.sqrt(prevX * prevX + prevY * prevY + prevZ * prevZ)
    if pmag > 0 and mag > 0 then
        local pdx = prevX / pmag
        local pdy = prevY / pmag
        local pdz = prevZ / pmag
        local ndx = cx / mag
        local ndy = cy / mag
        local ndz = cz / mag
        local dirDot = pdx * ndx + pdy * ndy + pdz * ndz
        score = score + (1 - math.max(dirDot, 0)) * 55
    end
    return score
end
function buildSeedRoutes(seedStr, mode, count, spread)
    local s = hashSeedString(seedStr, mode)
    for _ = 1, 16 do s = lcg(s) end
    local n = math.max(count or 128, 16)
    local seedStepMin = 17500000
    local seedStepSpan = 1000000
    local candMul = math.max(config.randomizerCandMul or 6, 2)
    local route = {}
    local startState, sx, sy, sz = lcgSequence(s)
    s = startState
    local prevX = (sx - 0.5) * seedStepMin * 0.1
    local prevY = (sy - 0.5) * seedStepMin * 0.1
    local prevZ = (sz - 0.5) * seedStepMin * 0.1
    local seedVelX, seedVelY, seedVelZ = 0, 0, 0
    for i = 1, n do
        local bestScore = -1e18
        local bestX, bestY, bestZ = prevX, prevY + seedStepMin * 0.5, prevZ
        local seedPredT = 0.008 + lcgFloat(s) * (0.08 - 0.008)
        local predX = prevX + seedVelX * seedPredT
        local predY = prevY + seedVelY * seedPredT
        local predZ = prevZ + seedVelZ * seedPredT
        for _ = 1, candMul do
            local ns, f1, f2, f3, _ = lcgSequence(s)
            s = ns
            local dist = seedStepMin + lcgFloat(ns) * seedStepSpan
            local dx = f1 * 2 - 1
            local dy = f2 * 2 - 1
            local dz = f3 * 2 - 1
            local dlen = math.sqrt(dx * dx + dy * dy + dz * dz)
            if dlen < 1e-6 then dlen = 1 end
            local cx = prevX + (dx / dlen) * dist
            local cy = prevY + (dy / dlen) * dist
            local cz = prevZ + (dz / dlen) * dist
            local scoreValue = scoreRouteCandidate(cx, cy, cz, prevX, prevY, prevZ, route, i - 1, predX, predY, predZ)
            if scoreValue > bestScore then
                bestScore = scoreValue
                bestX = cx
                bestY = cy
                bestZ = cz
            end
        end
        seedVelX = seedVelX + (bestX - prevX - seedVelX) * 0.25
        seedVelY = seedVelY + (bestY - prevY - seedVelY) * 0.25
        seedVelZ = seedVelZ + (bestZ - prevZ - seedVelZ) * 0.25
        route[i] = { bestX, bestY, bestZ }
        prevX = bestX
        prevY = bestY
        prevZ = bestZ
    end
    local out = {}
    for i = 1, n do
        out[i] = Vector3.new(route[i][1], route[i][2], route[i][3])
    end
    return out
end

function getGlueCFrame(targetCharacter, anchorPart, useHitbox, extraY, extraZ)
    local fighter = nil
    pcall(function()
        local controller = requireOptionalModule(player.PlayerScripts.Controllers.FighterController)
        if controller and targetCharacter then
            fighter = manipulation.findFighter(controller, targetCharacter)
        end
    end)
    local above = manipulation.isAbove(fighter)
    local head = anchorPart
    if useHitbox then
        head = manipulation.resolveHitbox(anchorPart, targetCharacter, fighter)
    end
    if not head then
        return nil
    end
    local okCf, headCFrame = pcall(function() return head.CFrame end)
    if not okCf or not headCFrame or not validVector(headCFrame.Position) then
        return nil
    end
    local baseOffset = above and Vector3.new(0, -0.7, 0.05) or Vector3.new(0, -3.85, 0.05)
    local offset = baseOffset + Vector3.new(0, extraY or 0, extraZ or 0)
    if above then
        return headCFrame + offset
    end
    local headPosition = headCFrame.Position
    local fromPosition = headPosition + offset
    local forward = Vector3.new(headPosition.X - fromPosition.X, 0, headPosition.Z - fromPosition.Z)
    if forward.Magnitude < 0.001 then
        forward = Vector3.new(0, 0, -1)
    end
    return CFrame.lookAt(fromPosition, fromPosition + forward, Vector3.new(0, -1, 0))
end

function buildTeleportationConstraint(playerRootPart, targetPart, overrideCFrame)
    destroyTeleportationConstraint()
    local okParent, targetCharacter = pcall(function() return targetPart.Parent end)
    if not okParent then
        return
    end
    local useHitbox = (config.teleportationAttachMode or "Head") == "Head"
    local snapCFrame = overrideCFrame or getGlueCFrame(targetCharacter, targetPart, useHitbox, config.teleportationHeight or 0, config.teleportationDist or 0)
    if not snapCFrame or not validVector(snapCFrame.Position) then
        return
    end
    local snaps = math.clamp(config.teleportationSnapCount or 3, 1, 8)
    if config.teleportationNonClientSided then
        getgenv().ncsCapture(playerRootPart, true)
        for _ = 1, snaps do
            setCFrame(playerRootPart, snapCFrame, true)
        end
        getgenv().ncsMark(playerRootPart)
    else
        for _ = 1, snaps do
            setCFrame(playerRootPart, snapCFrame)
        end
    end
    teleportationBoundRootPart = playerRootPart
    teleportationBoundTarget = targetPart
end
function getTeleportationAttachPart(char)
    if not char then
        return
    end
    local mode = config.teleportationAttachMode or "Head"
    if mode == "HRP" then
        return char:FindFirstChild("HumanoidRootPart")
    end
    local partRef = char:FindFirstChild(mode)
    if partRef then
        return partRef
    end
    return char:FindFirstChild("HumanoidRootPart")

function resolveTeleportationTarget()
    local currentTime = tick()
    local teamCheck = config.teleportationTeamCheck ~= false
    local cacheTtl = (teleportationTargetCache ~= nil) and 0.033 or 0
    if not teleportationTargetDirty and (currentTime - getgenv().teleportationTargetLast) < cacheTtl then
        if teleportationTargetCache and getgenv().teleportationIsValid(teleportationTargetCache, teamCheck) then
            return teleportationTargetCache, teleportationTargetRootPart, teleportationTargetHead
        end
    end
    teleportationTargetDirty = false
    getgenv().teleportationTargetLast = currentTime
    local playerRootPart = getgenv().getHumanoidRootPart(player)
    if not playerRootPart then
        teleportationTargetCache = nil
        teleportationTargetRootPart = nil
        teleportationTargetHead = nil
        return nil
    end
    local teleportationTargetName = config.teleportationTargetPlayerName
    local named = teleportationTargetName ~= nil and teleportationTargetName ~= "" and teleportationTargetName ~= "Closest"
    local picked, pickedHumanoidRootPart = nil, nil
    if named then
        local callSucceeded, sHumanoidRootPart = getgenv().teleportationIsValid(players:FindFirstChild(teleportationTargetName), teamCheck)
        if callSucceeded then
            picked = players:FindFirstChild(teleportationTargetName)
            pickedHumanoidRootPart = sHumanoidRootPart
        end
    end
    if not picked then
        local mode = config.teleportationPriority or "Closest"
        local playerPosition = playerRootPart.Position
        local bestS = math.huge
        for _, p in ipairs(players:GetPlayers()) do
            local callSucceeded, targetRootPart, targetHumanoid = getgenv().teleportationIsValid(p, teamCheck)
            if callSucceeded then
                local scoreValue = getgenv().teleportationScore(targetRootPart, targetHumanoid, playerPosition, mode)
                if scoreValue < bestS then
                    bestS = scoreValue
                    picked = p
                    pickedHumanoidRootPart = targetRootPart
                end
            end
        end
    end
    teleportationTargetCache = picked
    teleportationTargetRootPart = pickedHumanoidRootPart
    teleportationTargetHead = picked and getTeleportationAttachPart(picked.Character) or nil
    return teleportationTargetCache, teleportationTargetRootPart, teleportationTargetHead
end

    if teleportationActive then
        return
    end
    teleportationActive = true
    config.teleportation = true
    if config.teleportationNonClientSided then
        bindTeleportationRestore()
    end
    teleportationTargetCache = nil
    teleportationTargetRootPart = nil
    teleportationTargetHead = nil
    teleportationTargetDirty = true
    for _, p in ipairs(players:GetPlayers()) do
        local c = p.CharacterAdded:Connect(function() teleportationTargetDirty = true end)
        local characterRemovingConnection = p.CharacterRemoving:Connect(function() teleportationTargetDirty = true end)
        table.insert(teleportationConnections, c)
        table.insert(teleportationConnections, characterRemovingConnection)
    end
    local teleportationPlayerAddedConnection = players.PlayerAdded:Connect(function(p)
        teleportationTargetDirty = true
        local c = p.CharacterAdded:Connect(function() teleportationTargetDirty = true end)
        local characterRemovingConnection = p.CharacterRemoving:Connect(function() teleportationTargetDirty = true end)
        table.insert(teleportationConnections, c)
        table.insert(teleportationConnections, characterRemovingConnection)
    end)
    local teleportationPlayerRemovingConnection = players.PlayerRemoving:Connect(function() teleportationTargetDirty = true end)
    table.insert(teleportationConnections, teleportationPlayerAddedConnection)
    table.insert(teleportationConnections, teleportationPlayerRemovingConnection)
    local teleportationHbLast = 0
    teleportationConnection = runService.Heartbeat:Connect(function()
        if not config.teleportation then
            return
        end
        local currentTime = tick()
        if currentTime - teleportationHbLast < 0.016 then
            return
        end
        teleportationHbLast = currentTime
        local playerRootPart = getgenv().getHumanoidRootPart(player)
        if not playerRootPart then
            return
        end
        local characterHumanoid = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
        if not characterHumanoid or characterHumanoid.Health <= 0 then
            destroyTeleportationConstraint()
            return
        end
        local _, targetRootPart, head = resolveTeleportationTarget()
        local targetPart = head or targetRootPart
        if not targetPart then
            destroyTeleportationConstraint()
            return
        end
        pcall(buildTeleportationConstraint, playerRootPart, targetPart)
    end)
end

    local function outOfBounds(pos)

    local function spatialTick(target)
        local now = os.clock()
        local key = target.fighter or target
        local measurement = measurements[key]
        if measurement == nil then
            measurement = { ExpectedDuration = 1 }
            measurements[key] = measurement
        end
        local entryTime = measurement.LimitEntryTime
        if not outOfBounds(target.rootPart.Position) then
            if entryTime ~= nil then
                if equippedHasAmmo(target) then
                    measurement.ExpectedDuration = now - entryTime
                end
                measurement.LimitEntryTime = nil
            end
            return false
        end
        if entryTime == nil then
            measurement.LimitEntryTime = now
            entryTime = now
        end
        if equippedHasAmmo(target) then
            if measurement.ExpectedDuration - cfg.stability <= now - entryTime then
                return false
            end
        end
        return true
    end

    local function newShootLock()
        local lock = { lockedUntil = nil }
        function lock.shouldFire(shouldLock, duration)
            local now = os.clock()
            local locked = lock.lockedUntil ~= nil and now < lock.lockedUntil
            if shouldLock then
                lock.lockedUntil = now + duration
            end
            return locked or shouldLock
        end
        function lock.reset()
            lock.lockedUntil = nil
        end
        return lock
    end

    local function lookAtFlat(from, to)
        local n = to - from
        local forward = Vector3.new(n.X, 0, n.Z)
        if forward.Magnitude < 0.001 then

        return CFrame.lookAt(from, from + forward, Vector3.new(0, -1, 0))

    local function glueCFrame(target, above)
        local head = target.hitboxHead
        local offset = above and OFFSET_ABOVE or OFFSET_BELOW
        local glued = head.CFrame
        if above then
            return glued + offset
        end
        return lookAtFlat(glued.Position + offset, head.Position)

    local function randomEvasionCFrame(anchorPos, minRadius, maxRadius)
        local angle = randomGen:NextNumber(0, TAU)
        local radius = randomGen:NextNumber(minRadius, maxRadius)
        local base = CFrame.new(anchorPos + Vector3.new(math.cos(angle) * radius, 0, math.sin(angle) * radius))
        return base * CFrame.fromOrientation(
            randomGen:NextNumber(0, TAU),
            randomGen:NextNumber(0, TAU),
            randomGen:NextNumber(0, TAU)
        )
    end
    local function randomEvasion(rootCFrame)
        local baseRadius = cfg.baseRadius
        local maxRadius = baseRadius + baseRadius * cfg.radiusFactor
        local pos = rootCFrame.Position
        local anchor = cfg.anchorFromCharacter and pos or Vector3.new(0, pos.Y, 0)
        local targetCFrame = randomEvasionCFrame(anchor, baseRadius, maxRadius)
        local targetPos = targetCFrame.Position
        local x, y, z = targetPos.X, targetPos.Y, targetPos.Z
        local axis = randomGen:NextInteger(1, 3)
        if axis == 1 then
            x = FAR
        elseif axis == 2 then
            y = FAR
        else
            z = FAR
        end
        return targetCFrame - targetPos + Vector3.new(x, y, z)
    end
    local function evadePlan(rootCFrame)

    local function planMelee(dt, target, item, canFire)
        local head = target.hitboxHead
        local above = manipulation.isAbove(target.fighter)
        local aimCFrame = glueCFrame(target, above)
        local pitch, yaw, roll = target.rootPart.CFrame:ToOrientation()
        local n33 = above and -HALF_PI or HALF_PI
        local encA = manipulation.patchEncoding(above and manipulation.msEncA or manipulation.msEncC, n33, yaw, roll)
        local encB = manipulation.patchEncoding(above and manipulation.msEncB or manipulation.msEncD, n33, yaw, roll)
        local viewAngles = { Pitch = math.deg(pitch), Yaw = math.deg(yaw) }
        local objectId = manipulation.itemObjectId(item)
        local now = os.clock()
        if now < meleeWindowUntil then
            return aimCFrame, viewAngles, function()
                fireMelee("StartAiming", "HeavyAttackAnimation1", objectId, encA, encB, head, manipulation.msRot)
            end
        end
        if not meleeLock.shouldFire(canFire, dt * cfg.shootFrames) then
            return CFrame.new(math.random(-10000000, -100000), math.random(5000, 10000), math.random(-10000000, -100000)), nil, nil
        end
        if now < meleeCooldown then
            return CFrame.new(math.random(-10000000, -100000), math.random(5000, 10000), math.random(-10000000, -100000)), nil, nil
        end
        if manipulation.itemName(item) == "Knife" then
            meleeWindowUntil = now + 0.625
            meleeCooldown = now + 1.25
            return aimCFrame, viewAngles, function()
                fireMelee("StartAiming", "HeavyAttackAnimation1", objectId, encA, encB, head, manipulation.msRot)
            end
        end
        return aimCFrame, nil, function()
            fireMelee("StartShooting", "AttackAnimation1", objectId, encA, encB, head, manipulation.msRot)
        end
    end

    local function makePlan(dt, fighter, action, target, rootCFrame)
        if farmMode then
            return evadePlan(rootCFrame)
        end
        local canFire = target ~= nil and not spatialTick(target)
        if action == nil then
            return evadePlan(rootCFrame)
        end
        if action.type == "Swap" then
            local plan = evadePlan(rootCFrame)
            local item = action.item
            local index = action.index
            plan.action = function() equipItem(item, index) end
            return plan
        end
        if target == nil then
            return evadePlan(rootCFrame)
        end
        if action.itemType == "Melee" then
            local aimCFrame, angles, act = planMelee(dt, target, action.item, canFire)
            return { cframe = aimCFrame, viewAngles = angles, action = act, skipDefense = true, forceCrouch = true }
        end
        return {}
    end

    local lastDefensiveAngles = nil
    local function applyPlan(plan, target, root)
        local cf = plan.cframe
        if cf == nil or target == nil or plan.skipDefense then
            applyServerCFrame(root, cf)
            sendViewAngles(20, plan.viewAngles)
            return
        end
        applyServerCFrame(root, cf)
        if plan.aimPose or plan.defendInPlace then
            lastDefensiveAngles = nil
        end
        sendViewAngles(9, plan.viewAngles or lastDefensiveAngles)
    end
    local function resetState()
        meleeModeState.lastTargetWorld = nil
        lastDefensiveAngles = nil
        applyForcedCrouch(false)
        meleeWindowUntil = -1
        meleeCooldown = -1
        meleeLock.reset()
        renderRestore()
        sendViewAngles(20, nil)
    end

    local lastUpdate = 0
    meleeModeState.conn = runService.Heartbeat:Connect(function(dt)
        if not meleeModeState.active then
            return
        end
        local currentTime = tick()
        if currentTime - lastUpdate < 0.016 then
            return
        end
        lastUpdate = currentTime
        local fighter = getLocalFighter()
        local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
        if fighter == nil or root == nil then
            resetState()
            return
        end
        local rootCFrame = root.CFrame
        if config.meleeManipulation then
            getgenv().internalMeleeManipulationRealCFrame = rootCFrame
        end
        local action = getAction(fighter)
        local target = selectTarget()
        meleeModeState.lastTargetWorld = target ~= nil and target.rootPart.Position or nil
        local plan = makePlan(dt, fighter, action, target, rootCFrame)
        applyPlan(plan, target, root)
        applyForcedCrouch(plan.forceCrouch == true)
        if plan.action ~= nil then
            plan.action()
        end
    end)

    meleeModeState.cleanup = function()
        meleeModeState.conn = disconnect(meleeModeState.conn)
        pcall(function() runService:UnbindFromRenderStep(getgenv().meleeModeRenderId) end)
        renderRestore()
        applyForcedCrouch(false)
        meleeModeState.lastTargetWorld = nil
        pcall(function() workspace.FallenPartsDestroyHeight = originalFallenHeight end)
    end
end
getgenv().stopMeleeMode = function()
    meleeModeState.active = false
    meleeModeState.conn = disconnect(meleeModeState.conn)
    local cleanup = meleeModeState.cleanup
    meleeModeState.cleanup = nil
    if cleanup then
        pcall(cleanup)
    end
end

function antiSubspaceFriendly(mine)
    if not mine then
        return true
    end
    local creator = mine:GetAttribute("Creator") or mine:FindFirstChild("Creator")
    if creator and tostring(creator.Value or creator) == player.Name then
        return true
    end
    return false
end
function antiSubspaceNameMatch(o)
    local n = string.lower(o.Name)
    return string.find(n, "tripmine", 1, true) ~= nil or string.find(n, "subspace", 1, true) ~= nil
end
function antiSubspaceIsMine(o)
    if not (o:IsA("BasePart") or o:IsA("Model")) then
        return false
    end
    if not antiSubspaceNameMatch(o) then
        return false
    end
    local parent = o.Parent
    if parent and parent:IsA("Model") and antiSubspaceNameMatch(parent) then
        return false
    end
    return true
end
function antiSubspaceNeutralize(mine)
    pcall(function()
        if mine:IsA("BasePart") then
            mine.CanTouch = false
            mine.CanCollide = false
        end
        for _, d in ipairs(mine:GetDescendants()) do
            if d:IsA("BasePart") then
                d.CanTouch = false
                d.CanCollide = false
            end
        end
        mine:Destroy()
    end)
end
function antiSubspaceCheck(o)
    if not config.antiSubspace then
        return
    end
    if not o or not o.Parent then
        return
    end
    if not antiSubspaceIsMine(o) then
        return
    end
    if antiSubspaceFriendly(o) then
        return
    end
    antiSubspaceNeutralize(o)
end

function startTranslocation()
    if getgenv().transloc then
        return
    end
    getgenv().transloc = true
    config.translocation = true
    savePos()
    transConnection = disconnect(transConnection)
    local translocationCollectionService = cloneref(game:GetService("CollectionService"))
    local translocationOffset = -5
    local translocationLastUpdate = 0
    transConnection = runService.Heartbeat:Connect(function()
        if not config.translocation then
            return
        end
        local currentTime = tick()
        if currentTime - translocationLastUpdate < 0.1 then
            return
        end
        translocationLastUpdate = currentTime
        local playerRootPart = getCharacterInfo(player)
        if not playerRootPart then
            return
        end
        local currentCFrame = playerRootPart.CFrame
        local destCFrame = nil
        local foundPart = nil
        for _, v in pairs(translocationCollectionService:GetTagged("OutOfBoundsPart")) do
            if v:GetAttribute("KillDelay") == 0 then
                foundPart = v
                break
            end
        end
        if foundPart then
            destCFrame = foundPart.CFrame * CFrame.new(0, -foundPart.Size.Y / 2 + translocationOffset, 0)
        else
            local baseRad = config.evasionRadiusMin or 100
            local maxRad = config.evasionRadiusMax or 150
            local n = baseRad + baseRad * ((maxRad / baseRad) - 1)
            local ring = evasionRingPoint(currentCFrame.Position, baseRad, n)
            local ringPosition = ring.Position
            local x2, y2, z = ringPosition.X, ringPosition.Y, ringPosition.Z
            local pick = evasionRand:NextInteger(1, 3)
            if pick == 1 then x2 = 1073741824
            elseif pick == 2 then y2 = 1073741824
            else z = 1073741824 end
            destCFrame = CFrame.new(Vector3.new(x2, y2, z))
        end
        local destPos = destCFrame.Position
        if not validVector(destPos) then
            return
        end
        setVelocity(playerRootPart)
        setCFrame(playerRootPart, CFrame.new(destPos))
        setVelocity(playerRootPart)
        pcall(function()
            setCFrame(playerRootPart, CFrame.new(destPos), true)
        end)
    end)
end
function stopTranslocation()
    getgenv().transloc = false
    config.translocation = false
    transConnection = disconnect(transConnection)
    loadPos()
end

getgenv().startUnderground = function()
    if undergroundActive then
        return
    end
    undergroundActive = true
    config.underground = true
    lockY = nil
    lastPos = nil
    if not undergroundRotHooked then
        pcall(function()
            local mechCtrl = require(player.PlayerScripts:FindFirstChild("MechanicsController", true))
            undergroundRotFunction = mechCtrl._UpdateCharacterRotation
            if undergroundRotFunction then
                undergroundRotOriginal = hookfunction(undergroundRotFunction, function(self, ...)
                    if config.underground then
                        if not self.IsSliding and self:IsAlive() then
                            local fightObject = self.LocalFighter
                            local entity = fightObject and fightObject.Entity
                            if entity then
                                local rootPart = entity.RootPart
                                local humanoid = entity.Humanoid
                                if rootPart and humanoid then
                                    humanoid.AutoRotate = false
                                    local _, targetCharacterModel, _, firePart = findCombatTarget()
                                    if firePart and targetCharacterModel then
                                        local upTarget = Vector3.new(rootPart.Position.X, rootPart.Position.Y + 999999, rootPart.Position.Z)
                                        rootPart.CFrame = CFrame.new(rootPart.Position, upTarget)
                                        manipulation.fireToward(firePart, targetCharacterModel)
                                    end
                                end
                            end
                        end
                        return
                    end
                    return undergroundRotOriginal(self, ...)
                end)
                undergroundRotHooked = true
            end
        end)
    end
    undergroundConnection = runService.Heartbeat:Connect(function()
        if not config.underground then
            return
        end
        local character = player.Character
        if character and character:FindFirstChild("HumanoidRootPart") and character:FindFirstChild("Humanoid") then
            local root = character.HumanoidRootPart
            local humanoid = character.Humanoid
            local currentPosition = root.Position
            if not lockY then
                lockY = currentPosition.Y - 7
            end
            local moveDir = humanoid.MoveDirection
            if moveDir.Magnitude > 0 then
                local speed = humanoid.WalkSpeed
                lastPos = root.Position + (moveDir * speed * 0.016)
                setCFrame(root, CFrame.new(lastPos.X, lockY, lastPos.Z) * (root.CFrame - currentPosition))
            else
                if not lastPos then
                    lastPos = currentPosition
                end

local function rbGetField(obj, keys)
    if not obj then return nil end
    for _, k in ipairs(keys) do
        local v
        if pcall(function() v = obj[k] end) and v ~= nil then return v end
        v = nil
        if pcall(function() v = obj:Get(k) end) and v ~= nil then return v end
    end
    return nil
end

local function rbFieldExists(obj, key)
    local found = false
    pcall(function() if obj[key] ~= nil then found = true end end)
    if found then return true end
    pcall(function() if obj:Get(key) ~= nil then found = true end end)
    return found
end

local function rbForceField(obj, key, value)
    if not obj then return end
    if not rbFieldExists(obj, key) then return end
    pcall(function() obj[key] = value end)
    pcall(function() obj:Set(key, value) end)
end

local function rbIsReloading(item)
    local v = rbGetField(item, RB_RELOAD_KEYS)
    if v == true then return true end
    if type(v) == "number" and v > tick() then return true end
    return false
end

    local v = rbGetField(item, RB_COOLDOWN_KEYS)

local function rbChooseSlot(startIndex)
    for index = (startIndex or 1), #rbConfig.Priority do
        local name = rbConfig.Priority[index]
        local slot = rbSlots[name]
        if slot then
            local item = rbSlotItems[slot]
            if item == nil then return slot end
            if not rbIsReloading(item) then
                local ammo = rbGetAmmo(item)
                local dry = (ammo ~= nil and ammo <= 0)
                local threshold = (index == 1) and rbConfig.MinSwapCooldown or 0
                local cooling = rbGetCooldownRemaining(item) > threshold
                if not dry and not cooling then return slot end
            end
        end
    end
    return nil
end

local function rbUpdateFighter(fighterObj)
    local p = fighterObj.Player
    if not p then return end
    if not fighterObj.Entity or not fighterObj.Entity:IsAlive() or fighterObj:Get("IsSpectating") then
        rbDeflecting[p] = false
        return
    end
    local equipped = fighterObj.EquippedItem
    local isKatana = equipped and equipped.ViewModel and equipped.ViewModel.Name == "Katana"
    rbDeflecting[p] = (isKatana and (equipped._attack_cooldown or 0) > tick()) or false
end

local function rbUpdateFighters()
    if not rbFighterController or not rbFighterController.Objects then return end
    for _, fighterObj in pairs(rbFighterController.Objects) do
        rbUpdateFighter(fighterObj)
    end
end

local function rbIsEnemy(p)
    if p == rbLplr then return false end
    if not rbSpectateController then
        local pTeam = p:GetAttribute("TeamID")
        local lTeam = rbLplr:GetAttribute("TeamID")
        if pTeam and lTeam then return pTeam ~= lTeam end
        return true
    end
    local duel = rbSpectateController.CurrentDuelSubject
    local localDueler = duel and duel:GetDueler(rbLplr)
    local localTeam = localDueler and localDueler:Get("TeamID") or nil
    if localTeam and duel and duel.Duelers then
        for _, dueler in pairs(duel.Duelers) do
            if dueler.Player == p then
                return dueler:Get("TeamID") ~= localTeam
            end
        end
    end
    local pTeam = p:GetAttribute("TeamID")
    local lTeam = rbLplr:GetAttribute("TeamID")
    if pTeam and lTeam then return pTeam ~= lTeam end
    return true
end


local function rbTrackRoundStart()
    if not rbSpectateController then return end
    local duel = rbSpectateController.CurrentDuelSubject
    if duel ~= rbLastDuel then
        rbLastDuel = duel
    end
end


local function rbResetWeapons()
    rbSlotItems = {}
    rbCurrentSlot = nil
    rbSwitchAt = 0
    rbFallbackSince = 0
    rbFallbackTarget = nil
    rbLastPrimaryDraw = 0
    rbGunDry = false
    rbDeflecting = {}
end


local function rbIsNaN(v) return v ~= v end
local function rbIsInf(v) return math.abs(v) == math.huge end
local function rbFinite(v) return not rbIsNaN(v) and not rbIsInf(v) end
local function rbValidVec(v) return v and rbFinite(v.X) and rbFinite(v.Y) and rbFinite(v.Z) end

local function rbFindFighter(controller, targetCharacter)
    if not controller or not targetCharacter then return nil end
    local okObjects, objects = pcall(function() return controller.Objects end)
    if not okObjects or type(objects) ~= "table" then return nil end
    local owner = rbPlrs:GetPlayerFromCharacter(targetCharacter)
    for _, fighter in ipairs(objects) do
        local okMatch, matched = pcall(function()
            if owner ~= nil and rawget(fighter, "Player") == owner then return true end
            local entity = rawget(fighter, "Entity")
            local model = entity and rawget(entity, "Model")
            return model ~= nil and model == targetCharacter
        end)
        if okMatch and matched then return fighter end
    end
    return nil
end

local function rbIsAbove(fighter)
    if not fighter then return true end
    local okObs, obs = pcall(function()
        return rawget(fighter, "itemObserver") or rawget(fighter, "ItemObserver")
    end)
    if not okObs or not obs then return true end
    local ok, result = pcall(function()
        local equipped = obs:GetEquippedItem()
        if equipped ~= nil and equipped.ViewModel and equipped.ViewModel.Name == "Katana" then
            local pitch = math.deg(fighter:GetCameraRotation().X)
            if 22 < pitch and pitch < 91 then return "Below" end
            return "Above"
        end
        for _, it in pairs(obs:GetItems()) do
            if it.name == "Riot Shield" then
                local pitch = math.deg(fighter:GetCameraRotation().X)
                if 315 < pitch and pitch < 360 or 0 < pitch and pitch < 91 then return "Above" end
                return "Below"
            end
        end
        return "None"
    end)
    return not (ok and result == "Below")
end

local function rbResolveHitbox(targetPart, targetCharacter, fighter)
    local hitbox = nil
    if targetCharacter then hitbox = targetCharacter:FindFirstChild("HitboxHead") end
    if not hitbox and fighter then
        local okModel, model = pcall(function() return rawget(rawget(fighter, "Entity"), "Model") end)
        if okModel and model then hitbox = model:FindFirstChild("HitboxHead") end
    end
    return hitbox or targetPart
end

local function rbTpIsValid(p, teamCheck, keepFf)
    if not p or p == rbLplr then return false end
    local ch = p.Character
    if not ch then return false end
    local targetRootPart = ch:FindFirstChild("HumanoidRootPart")
    local targetHumanoid = ch:FindFirstChildOfClass("Humanoid")
    if not targetRootPart or not targetHumanoid or targetHumanoid.Health <= 0 then return false end
    if not keepFf and ch:FindFirstChildOfClass("ForceField") then return false end
    if teamCheck and not rbIsEnemy(p) then return false end
    return true, targetRootPart, targetHumanoid
end

local function rbTpScore(targetRootPart, targetHumanoid, playerPosition, mode)
    local distanceValue = (targetRootPart.Position - playerPosition).Magnitude
    if mode == "Lowest Health" then return targetHumanoid.Health + distanceValue * 0.000001 end
    if mode == "Closest To Cursor" then
        local cam = rbWs.CurrentCamera
        if cam then
            local v, on = cam:WorldToViewportPoint(targetRootPart.Position)
            local mouseLocation = rbUis:GetMouseLocation()
            local screenDistance = (Vector2.new(v.X, v.Y) - mouseLocation).Magnitude
            if not on then screenDistance = screenDistance + 100000 end
            return screenDistance + distanceValue * 0.000001
        end
    end
    return distanceValue
end

local function rbGetGlueCFrame(targetCharacter, anchorPart, useHitbox, extraY, extraZ)
    local fighter = nil
    pcall(function()
        if rbFighterController and targetCharacter then
            fighter = rbFindFighter(rbFighterController, targetCharacter)
        end
    end)
    local above = rbIsAbove(fighter)
    local head = anchorPart
    if useHitbox then head = rbResolveHitbox(anchorPart, targetCharacter, fighter) end
    if not head then return nil end
    local okCf, headCFrame = pcall(function() return head.CFrame end)
    if not okCf or not headCFrame or not rbValidVec(headCFrame.Position) then return nil end
    local baseOffset = above and Vector3.new(0, -0.7, 0.05) or Vector3.new(0, -3.85, 0.05)
    local offset = baseOffset + Vector3.new(0, extraY or 0, extraZ or 0)
    if above then return headCFrame + offset end
    local headPosition = headCFrame.Position
    local fromPosition = headPosition + offset
    local forward = Vector3.new(headPosition.X - fromPosition.X, 0, headPosition.Z - fromPosition.Z)
    if forward.Magnitude < 0.001 then forward = Vector3.new(0, 0, -1) end
    return CFrame.lookAt(fromPosition, fromPosition + forward, Vector3.new(0, -1, 0))
end

local function rbBuildTpConstraint(playerRootPart, targetPart)
    rbDestroyTpConstraint()
    local okParent, targetCharacter = pcall(function() return targetPart.Parent end)
    if not okParent then return nil end
    local useHitbox = (rbConfig.TeleportationAttachMode or "Head") == "Head"
    local snapCFrame = rbGetGlueCFrame(targetCharacter, targetPart, useHitbox, rbConfig.TeleportationHeight or 0, rbConfig.TeleportationDist or 0)
    if not snapCFrame or not rbValidVec(snapCFrame.Position) then return nil end
    local snaps = math.clamp(rbConfig.TeleportationSnapCount or 3, 1, 8)
    if rbConfig.TeleportationNonClientSided then
        getgenv().ncsCapture(playerRootPart, true)
        for _ = 1, snaps do setCFrame(playerRootPart, snapCFrame, true) end
    else
        for _ = 1, snaps do setCFrame(playerRootPart, snapCFrame) end
    end
    if rbConfig.TeleportationNonClientSided then rbBindTpRestore() end
    return snapCFrame
end

local function rbGetTpAttachPart(char)
    if not char then return nil end
    local mode = rbConfig.TeleportationAttachMode or "Head"
    if mode == "HRP" then return char:FindFirstChild("HumanoidRootPart") end
    local partRef = char:FindFirstChild(mode)
    if partRef then return partRef end
    return char:FindFirstChild("HumanoidRootPart")
end

local function rbResolveTpTarget()
    local currentTime = tick()
    local teamCheck = rbConfig.TeleportationTeamCheck ~= false
    local cacheTtl = (rbTpTargetCache ~= nil) and 0.033 or 0
    if not rbTpTargetDirty and (currentTime - rbTpLastResolve) < cacheTtl then
        if rbTpTargetCache and rbTpIsValid(rbTpTargetCache, teamCheck, true) then
            return rbTpTargetCache, rbTpTargetRootPart, rbTpTargetHead
        end
    end
    rbTpTargetDirty = false
    rbTpLastResolve = currentTime
    local playerRootPart = rbLplr.Character and rbLplr.Character:FindFirstChild("HumanoidRootPart")
    if not playerRootPart then
        rbTpTargetCache = nil
        rbTpTargetRootPart = nil
        rbTpTargetHead = nil
        return nil
    end
    local targetName = rbConfig.TeleportationTargetPlayerName
    local named = targetName ~= nil and targetName ~= "" and targetName ~= "Closest"
    local picked, pickedRoot = nil, nil
    if named then
        local callSucceeded, sRoot = rbTpIsValid(rbPlrs:FindFirstChild(targetName), teamCheck, true)
        if callSucceeded then
            picked = rbPlrs:FindFirstChild(targetName)
            pickedRoot = sRoot
        end
    end
    if not picked then
        local mode = rbConfig.TeleportationPriority or "Closest"
        local playerPosition = playerRootPart.Position
        local bestS = math.huge
        for _, p in ipairs(rbPlrs:GetPlayers()) do
            local callSucceeded, tRoot, tHum = rbTpIsValid(p, teamCheck, true)
            if callSucceeded then
                local scoreValue = rbTpScore(tRoot, tHum, playerPosition, mode)
                if scoreValue < bestS then
                    bestS = scoreValue
                    picked = p
                    pickedRoot = tRoot
                end
            end
        end
    end
    rbTpTargetCache = picked
    rbTpTargetRootPart = pickedRoot
    rbTpTargetHead = picked and rbGetTpAttachPart(picked.Character) or nil
    return rbTpTargetCache, rbTpTargetRootPart, rbTpTargetHead
end

local function rbTeleportationStep()
    local playerRootPart = rbLplr.Character and rbLplr.Character:FindFirstChild("HumanoidRootPart")
    if not playerRootPart then return nil end
    local characterHumanoid = rbLplr.Character:FindFirstChildOfClass("Humanoid")
    if not characterHumanoid or characterHumanoid.Health <= 0 then
        rbDestroyTpConstraint()
        return nil
    end
    local targetPlayer, targetRootPart, attachPart = rbResolveTpTarget()
    local targetPart = attachPart or targetRootPart
    if not targetPlayer or not targetPart then
        rbDestroyTpConstraint()
        return nil
    end
    local okBuild, snapCFrame = pcall(rbBuildTpConstraint, playerRootPart, targetPart)
    local targetCharacter = targetPlayer.Character
    local targetHead = targetCharacter and targetCharacter:FindFirstChild("Head")
    return targetPlayer, targetRootPart, targetHead, (okBuild and snapCFrame or nil)
end

local function rbIsTargetInvincible(targetPlayer)
    if not targetPlayer or not targetPlayer.Character then return false end
    local character = targetPlayer.Character
    if character:FindFirstChildOfClass("ForceField") then return true end
    if not rbFighterController or not rbFighterController.Objects then return false end
    for _, fighterObj in pairs(rbFighterController.Objects) do
        local ok, matched = pcall(function()
            return rawget(fighterObj, "Player") == targetPlayer
        end)
        if ok and matched then
            local okData, data = pcall(rawget, fighterObj, "Data")
            if okData and type(data) == "table" and rawget(data, "IsInvincible") == true then
                return true
            end
            local okEntity, entity = pcall(rawget, fighterObj, "Entity")
            if okEntity and entity then
                local okEData, eData = pcall(rawget, entity, "Data")
                if okEData and type(eData) == "table" and rawget(eData, "IsInvincible") == true then
                    return true
                end
            end
            return false
        end
    end
    return false
end

getgenv().rbZombieTower = {
    FireRate = 0.0005,
    HoverHeight = 10,
    MeleeHoverHeight = 3,
    AutoCollect = true,
    CollectRadius = 120,
    CollectHold = 0.3,
    keywords = { gem = 3, chest = 2.5, loot = 2, coin = 1.5, disconnectall = 1 },
    keywordList = { "gem", "chest", "loot", "coin", "disconnectall" },
    restoreId = "__builtinRagebot_ztRender",
    lastFire = 0,
    lastEquip = 0,
    collectParts = {},
    collectUpdated = 0,
    collectTick = 0,
    collectUntil = 0,
}
do
    local zt = getgenv().rbZombieTower

    function zt.clearWeld()
        if zt.weld then
            pcall(function()
                zt.weld:Destroy()
            end)
        end
        zt.weld = nil
        zt.weldTarget = nil
        zt.weldHeight = nil
    end

    function zt.snapRoot(myRoot, cf)
        if rbConfig.TeleportationNonClientSided and zt.restoreCF == nil then
            zt.restoreRoot = myRoot
            zt.restoreCF = myRoot.CFrame
            zt.restoreVel = myRoot.AssemblyLinearVelocity
            zt.restoreRotVel = myRoot.AssemblyAngularVelocity
        end
        myRoot.CFrame = cf
        myRoot.AssemblyLinearVelocity = Vector3.zero
        myRoot.AssemblyAngularVelocity = Vector3.zero
    end

    function zt.renderRestore()
        if zt.restoreCF == nil then
            return
        end
        zt.clearWeld()
        local root = zt.restoreRoot
        if root and root.Parent then
            root.CFrame = zt.restoreCF
            root.AssemblyLinearVelocity = zt.restoreVel
            root.AssemblyAngularVelocity = zt.restoreRotVel
        end
        zt.restoreCF = nil
        zt.restoreRoot = nil
    end

    function zt.bind()
        pcall(function()
            rbRunS:UnbindFromRenderStep(zt.restoreId)
        end)
        pcall(function()
            rbRunS:BindToRenderStep(zt.restoreId, Enum.RenderPriority.First.Value, zt.renderRestore)
        end)
    end

    function zt.unbind()
        pcall(function()
            rbRunS:UnbindFromRenderStep(zt.restoreId)
        end)
        zt.renderRestore()
        zt.clearWeld()
        zt.npcConn = disconnect(zt.npcConn)
        zt.collectConn = disconnect(zt.collectConn)
        zt.npcs = {}
        zt.collectParts = {}
    end

    function zt.onModeChanged(zombie)
        if not rbHeartbeatConn then
            return
        end
        rbResetWeapons()
        rbTpTargetDirty = true
        if zombie then
            rbDestroyTpConstraint()
            rbUnbindTpRestore()
            zt.bind()
        else
            zt.unbind()
            if rbTpActive and rbConfig.TeleportationNonClientSided then
                rbBindTpRestore()
            end
        end
    end

    function zt.order()
        local list = {}
        if rbConfig.ZtPrimary then
            list[#list + 1] = rbSlots.Primary
        end
        if rbConfig.ZtMelee then
            list[#list + 1] = rbSlots.Secondary
        end
        return list
    end

    function zt.chooseSlot(order)
        for _, slot in ipairs(order) do
            local item = rbSlotItems[slot]
            if item == nil then
                return slot
            end
            if not rbIsReloading(item) then
                local ammo = rbGetAmmo(item)
                if ammo == nil or ammo > 0 then
                    return slot
                end
            end
        end
        return order[1]
    end

    function zt.hoverHeight(slot)
        if slot == rbSlots.Secondary then
            return zt.MeleeHoverHeight
        end
        return zt.HoverHeight
    end

    zt.npcs = {}
    function zt.trackNpc(inst)
        if inst:IsA("Humanoid") then
            local model = inst.Parent
            if model and model:IsA("Model") and not rbPlrs:GetPlayerFromCharacter(model) then
                zt.npcs[model] = true
            end
        end
    end

    function zt.getTarget()
        local char = rbLplr.Character
        if not char then
            return nil, nil, nil
        end
        local myRoot = char:FindFirstChild("HumanoidRootPart")
        if not myRoot then
            return nil, nil, nil
        end
        if not zt.npcConn then
            zt.npcs = {}
            zt.npcConn = rbWs.DescendantAdded:Connect(function(inst)
                pcall(zt.trackNpc, inst)
            end)
            for _, inst in ipairs(rbWs:GetDescendants()) do
                zt.trackNpc(inst)
            end
        end
        local closestModel = nil
        local closestRoot = nil
        local closestHead = nil
        local closestDist = 500
        for obj in pairs(zt.npcs) do
            if not obj.Parent then
                zt.npcs[obj] = nil
            else
                local pRoot = obj:FindFirstChild("HumanoidRootPart")
                local pHead = obj:FindFirstChild("Head")
                local pHum = obj:FindFirstChildWhichIsA("Humanoid")
                if pRoot and pHead and pHum and pHum.Health > 0 and obj ~= char then
                    local dist = (myRoot.Position - pRoot.Position).Magnitude
                    if dist < closestDist then
                        closestDist = dist
                        closestModel = obj
                        closestRoot = pRoot
                        closestHead = pHead
                    end
                end
            end
        end
        return closestModel, closestRoot, closestHead
    end

    function zt.weldAbove(myRoot, targetRoot, height)
        if zt.weld and zt.weld.Parent and zt.weldTarget == targetRoot and targetRoot.Parent and zt.weld.Part0 == myRoot and zt.weldHeight == height then
            return
        end
        zt.clearWeld()
        local abovePos = targetRoot.Position + Vector3.new(0, height, 0)
        zt.snapRoot(myRoot, CFrame.lookAt(abovePos, targetRoot.Position))
        local weld = Instance.new("WeldConstraint")
        weld.Part0 = myRoot
        weld.Part1 = targetRoot
        weld.Parent = myRoot
        zt.weld = weld
        zt.weldTarget = targetRoot
        zt.weldHeight = height
    end

    function zt.considerCollect(o)
        if o:IsA("BasePart") then
            local ownChar = rbLplr.Character
            if not (ownChar and o:IsDescendantOf(ownChar)) then
                local n = string.lower(o.Name)
                local itemVal = 0
                for _, kw in ipairs(zt.keywordList) do
                    if string.find(n, kw, 1, true) then
                        local v = zt.keywords[kw]
                        if v > itemVal then
                            itemVal = v
                        end
                    end
                end
                if itemVal > 0 then
                    zt.collectParts[#zt.collectParts + 1] = { partRef = o, secondaryValue = itemVal }
                end
            end
        end
    end

    function zt.refreshCollect(char)
        if not zt.collectConn then
            zt.collectParts = {}
            zt.collectConn = rbWs.DescendantAdded:Connect(function(o)
                pcall(zt.considerCollect, o)
            end)
            for _, o in ipairs(rbWs:GetDescendants()) do
                zt.considerCollect(o)
            end
        else
            local alive = {}
            for _, entry in ipairs(zt.collectParts) do
                if entry.partRef and entry.partRef.Parent then
                    alive[#alive + 1] = entry
                end
            end
            zt.collectParts = alive
        end
    end

    function zt.pickCollectible(myRoot)
        local best = nil
        local bestScore = math.huge
        local myPos = myRoot.Position
        for _, entry in ipairs(zt.collectParts) do
            local o = entry.partRef
            if o and o.Parent then
                local ok, opos = pcall(function()
                    return o.Position
                end)
                if ok and opos then
                    local dist = (myPos - opos).Magnitude
                    if dist < zt.CollectRadius then
                        local score = dist / entry.secondaryValue
                        if score < bestScore then
                            bestScore = score
                            best = o
                        end
                    end
                end
            end
        end
        return best
    end

    function zt.step()
        local char = rbLplr.Character
        local myRoot = char and char:FindFirstChild("HumanoidRootPart")
        if not myRoot or not rbConfig.Enabled then
            zt.clearWeld()
            return
        end
        local order = zt.order()
        if #order == 0 then
            zt.clearWeld()
            return
        end
        local now = tick()
        local want = zt.chooseSlot(order)
        local lf = rbFighterController and rbFighterController.LocalFighter
        if lf and (want ~= rbCurrentSlot or now - zt.lastEquip >= 1) then
            zt.lastEquip = now
            if rbEquipSlot(want) then
                rbCurrentSlot = want
            end
        end
        if zt.AutoCollect and now >= zt.collectUntil and now - zt.collectTick >= 0.5 then
            zt.collectTick = now
            if now - zt.collectUpdated >= 2 then
                zt.collectUpdated = now
                zt.refreshCollect(char)
            end
            local best = zt.pickCollectible(myRoot)
            if best then
                zt.clearWeld()
                zt.collectPart = best
                zt.collectUntil = now + zt.CollectHold
            end
        end
        if now < zt.collectUntil then
            if zt.collectPart and zt.collectPart.Parent then
                zt.snapRoot(myRoot, CFrame.new(zt.collectPart.Position))
            end
            return
        end
        local targetModel, targetRoot, targetHead = zt.getTarget()
        if not targetModel or not targetHead or not targetRoot then
            zt.clearWeld()
            return
        end
        local height = zt.hoverHeight(want)
        zt.weldAbove(myRoot, targetRoot, height)
        if not rbFighterController or not rbFighterController.LocalFighter then
            return
        end
        local item = rbFighterController.LocalFighter.EquippedItem
        if not item then
            return
        end
        if tick() - zt.lastFire < zt.FireRate then
            return
        end
        zt.lastFire = tick()
        local originPos = targetRoot.Position + Vector3.new(0, height, 0)
        local targetPos = targetHead.Position
        local aimCF = CFrame.lookAt(originPos, targetPos)
        local targetCF = targetHead.CFrame
        local randomOffset = Vector3.new((math.random() - 0.5) * 0.1, (math.random() - 0.5) * 0.1, (math.random() - 0.5) * 0.1)
        local aimedPos = targetPos + randomOffset
        local objSpaceHeadOffset = targetHead.CFrame:ToObjectSpace(CFrame.new(aimedPos))
        local cameradata = {}
        cameradata[utf8.char(1)] = {
            [utf8.char(0)] = rbUtil:EncodeCFrame(aimCF),
            [utf8.char(1)] = rbUtil:EncodeCFrame(targetCF),
            [utf8.char(2)] = targetHead,
            [utf8.char(3)] = rbUtil:EncodeCFrame(objSpaceHeadOffset)
        }
        rbRepS.Remotes.Replication.Fighter.UseItem:FireServer(
            item:Get("ObjectID"),
            rbEnum:ToEnum("StartShooting"),
            cameradata,
            nil
        )
    end
end

getgenv().startBuiltinRagebot = function()
    if rbHeartbeatConn then return end
    rbConfig.Enabled = true
    rbTpActive = true
    if rbConfig.TeleportationNonClientSided then rbBindTpRestore() end
    if rbConfig.Mode == "Zombie Tower" then getgenv().rbZombieTower.bind() end
    rbTpTargetCache = nil
    rbTpTargetRootPart = nil
    rbTpTargetHead = nil
    rbTpTargetDirty = true
    for _, p in ipairs(rbPlrs:GetPlayers()) do
        local c = p.CharacterAdded:Connect(function() rbTpTargetDirty = true end)
        local d = p.CharacterRemoving:Connect(function() rbTpTargetDirty = true end)
        table.insert(rbTpConns, c)
        table.insert(rbTpConns, d)
    end
    local padd = rbPlrs.PlayerAdded:Connect(function(p)
        rbTpTargetDirty = true
        local c = p.CharacterAdded:Connect(function() rbTpTargetDirty = true end)
        local d = p.CharacterRemoving:Connect(function() rbTpTargetDirty = true end)
        table.insert(rbTpConns, c)
        table.insert(rbTpConns, d)
    end)
    local prem = rbPlrs.PlayerRemoving:Connect(function() rbTpTargetDirty = true end)
    table.insert(rbTpConns, padd)
    table.insert(rbTpConns, prem)
    task.spawn(function()
        local lf = rbFighterController and rbFighterController.LocalFighter
        while not lf do
            task.wait(0.1)
            lf = rbFighterController and rbFighterController.LocalFighter
        end
        task.wait(0.3)
        if rbConfig.Mode == "Zombie Tower" then return end
        rbEquipSlot(rbSlots.Primary)
        task.wait(0.1)
        if lf.EquippedItem then rbSlotItems[rbSlots.Primary] = lf.EquippedItem end
        rbCurrentSlot = rbSlots.Primary
    end)
    rbLplr.CharacterAdded:Connect(function()
        rbResetWeapons()
        task.wait(0.5)
        if rbConfig.Mode == "Zombie Tower" then return end
        rbEquipSlot(rbSlots.Primary)
        rbCurrentSlot = rbSlots.Primary
    end)
    rbHeartbeatConn = rbRunS.Heartbeat:Connect(function()
        if rbConfig.Mode == "Zombie Tower" then
            getgenv().rbZombieTower.step()
            return
        end
        rbTrackRoundStart()
        rbUpdateFighters()
        if rbConfig.CooldownMode == "continuous" then rbApplyCooldownOverride() end

        local targetPlayer, targetRoot, targetHead
        local snapCFrame = nil
        targetPlayer, targetRoot, targetHead, snapCFrame = rbTeleportationStep()

        if rbConfig.Enabled then
            local primaryItem = rbSlotItems[rbSlots.Primary]
            if primaryItem then
                local ammo = rbGetAmmo(primaryItem)
                if ammo ~= nil then
                    if ammo <= 0 then rbGunDry = true end
                elseif rbIsReloading(primaryItem) then
                    rbGunDry = true
                end
            end
            local poking = (tick() - rbLastPrimaryDraw) < rbConfig.PokeHold
            local startIndex = 1
            if rbGunDry and rbConfig.StayOnFallback then startIndex = 2 end
            local want = rbChooseSlot(startIndex)
            if want and want ~= rbCurrentSlot and not poking then
                local onFallback = rbCurrentSlot ~= nil and rbSlotRank(rbCurrentSlot) > 1
                local upgrading = onFallback and rbSlotRank(want) < rbSlotRank(rbCurrentSlot)
                local sameTarget = (targetPlayer == rbFallbackTarget)
                local holdActive = sameTarget and (tick() - rbFallbackSince) < rbConfig.FallbackHold
                local blocked = upgrading and (rbConfig.StayOnFallback or holdActive)
                if not blocked then
                    if rbEquipSlot(want) then
                        rbCurrentSlot = want
                        rbSwitchAt = tick()
                        if rbSlotRank(want) > 1 then
                            rbFallbackSince = tick()
                            rbFallbackTarget = targetPlayer
                        end
                    end
                end
            end
            if rbConfig.PokePrimary and not rbConfig.StayOnFallback
                and rbCurrentSlot and rbSlotRank(rbCurrentSlot) > 1
                and (tick() - rbLastPrimaryDraw) >= rbConfig.PokeInterval then
                local p = rbSlotItems[rbSlots.Primary]
                local dry = p and ((rbGetAmmo(p) or 1) <= 0 or rbIsReloading(p))
                if dry then
                    rbLastPrimaryDraw = tick()
                    if rbEquipSlot(rbSlots.Primary) then
                        rbCurrentSlot = rbSlots.Primary
                        rbSwitchAt = tick()
                    end
                end
            end
        end

        if not rbConfig.Enabled then return end
        if tick() - rbSwitchAt < rbConfig.SwitchDelay then return end
        if not targetPlayer or not targetHead or not targetRoot then return end
        if rbDeflecting[targetPlayer] then return end
        if rbIsTargetInvincible(targetPlayer) then return end
        if not rbLplr.Character or not rbLplr.Character:FindFirstChild("HumanoidRootPart") then return end
        if not rbFighterController or not rbFighterController.LocalFighter then return end
        local item = rbFighterController.LocalFighter.EquippedItem
        if not item then return end
        local originPos = snapCFrame and snapCFrame.Position or targetRoot.Position
        local targetPos = targetHead.Position
        local aimCF = CFrame.lookAt(originPos, targetPos)
        local targetCF = targetHead.CFrame
        local randomOffset = Vector3.new((math.random() - 0.5) * 0.1, (math.random() - 0.5) * 0.1, (math.random() - 0.5) * 0.1)
        local aimedPos = targetPos + randomOffset
        local objSpaceHeadOffset = targetHead.CFrame:ToObjectSpace(CFrame.new(aimedPos))
        local cameradata = {}
        cameradata[utf8.char(1)] = {
            [utf8.char(0)] = rbUtil:EncodeCFrame(aimCF),
            [utf8.char(1)] = rbUtil:EncodeCFrame(targetCF),
            [utf8.char(2)] = targetHead,
            [utf8.char(3)] = rbUtil:EncodeCFrame(objSpaceHeadOffset)
        }
        rbRepS.Remotes.Replication.Fighter.UseItem:FireServer(
            item:Get("ObjectID"),
            rbEnum:ToEnum("StartShooting"),
            cameradata,
            nil
        )
        if rbConfig.CooldownMode == "onfire" then rbApplyCooldownOverride() end
    end)
    rbInputConn = rbUis.InputBegan:Connect(function(input, processed)
        if processed then return end
        if input.KeyCode == Enum.KeyCode.V then
            rbConfig.Enabled = not rbConfig.Enabled
            if ui.rbEnabledTog then
                ui.rbEnabledTog:SetValue(rbConfig.Enabled)
            end
        end
    end)
end

    coordinateSpoofConnection = disconnect(coordinateSpoofConnection)
    config.coordinateSpoof = true
    getgenv().ncsBind("coordinateSpoof")
    local coordinateSpoofLastUpdate = 0
    coordinateSpoofConnection = runService.Heartbeat:Connect(function()
        if not config.coordinateSpoof then
            return
        end
        local now = tick()
        if now - coordinateSpoofLastUpdate < (config.coordinateSpoofRate or 0.016) then
            return
        end
        coordinateSpoofLastUpdate = now
        local humanoidRootPart = getCharacterInfo(player)
        if not humanoidRootPart then
            return
        end
        local ncsEnv = getgenv()
        ncsEnv.ncsCapture(humanoidRootPart, true)
        local realCFrame = ncsEnv.ncsRestoreCFrame or humanoidRootPart.CFrame
        local intensity = config.coordinateSpoofIntensity or "Regular"
        local amount = config.coordinateSpoofAmount or 40
        local intensityMul = 1
        if intensity == "Strong" then intensityMul = 2.5
        elseif intensity == "Minimal" then intensityMul = 0.3 end
        local intensityValue = amount * intensityMul
        local iP = math.max(1, math.floor(intensityValue * 10))
        local iV = math.max(1, math.floor(intensityValue * 100))
        local realPosition = realCFrame.Position
        local fakePosition = Vector3.new(
            realPosition.X + math.random(-iP, iP),
            realPosition.Y + math.random(-iP, iP),
            realPosition.Z + math.random(-iP, iP)
        )
        local fakeVel = Vector3.new(
            math.random(-iV, iV),
            math.random(-iV, iV),
            math.random(-iV, iV)
        )
        if validVector(fakePosition) then
            setCFrame(humanoidRootPart, CFrame.new(fakePosition) * (realCFrame - realPosition), fakeVel, fakeVel)
        end
    end)
end

    local toggleRegistry = library.Toggles or {}
    local optionRegistry = library.Options or {}
    local function attach(element, handler, runAfter)
        if not element then
            return
        end
        local previous = element.Changed
        if previous == nil or type(previous) == "function" then
            element.Changed = function(...)
                if not runAfter then
                    handler(...)
                end
                if previous then
                    previous(...)
                end
                if runAfter then
                    handler(...)
                end
            end
        else
            element:OnChanged(handler)
        end
    end
    local function link(masterId, childIds, ...)
        local master = toggleRegistry[masterId]
        if not master then
            return
        end
        local extras = { ... }
        local extraCount = select("#", ...)
        local function apply(value)
            local visible = value == true
            for i = 1, #childIds do
                local child = toggleRegistry[childIds[i]] or optionRegistry[childIds[i]]
                if child then
                    pcall(child.SetVisible, child, visible)
                end
            end
            for i = 1, extraCount do
                local extra = extras[i]
                if extra then
                    pcall(extra.SetVisible, extra, visible)
                end
            end
        end
        attach(master, apply, false)
        apply(master.Value)
    end
