local function isPress(i) return i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch end
local function isMove(i) return i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch end

local function drag(handle, target, onTap, thresh)
    thresh = thresh or 0
    local down, moved, press, start, origin
    handle.InputBegan:Connect(function(i)
        if not isPress(i) then return end
        down, moved, press = true, false, i
        start, origin = Vector2.new(i.Position.X, i.Position.Y), target.AbsolutePosition + target.AbsoluteSize / 2
        local c
        c = i.Changed:Connect(function()
            if i.UserInputState ~= Enum.UserInputState.End then return end
            c:Disconnect()
            if down then down = false; if not moved and onTap then onTap() end end
        end)
    end)
    UIS.InputChanged:Connect(function(i)
        if not down then return end
        local touch = press.UserInputType == Enum.UserInputType.Touch
        if touch and i ~= press then return end
        if not touch and i.UserInputType ~= Enum.UserInputType.MouseMovement then return end
        local d = Vector2.new(i.Position.X, i.Position.Y) - start
        if not moved then if d.Magnitude < thresh then return end; moved = true end
        local vp, half = cam.ViewportSize, target.AbsoluteSize / 2
        local p = origin + d
        target.Position = UDim2.fromOffset(math.clamp(p.X, half.X, math.max(half.X, vp.X - half.X)), math.clamp(p.Y, half.Y, math.max(half.Y, vp.Y - half.Y)))
    end)
    UIS.WindowFocusReleased:Connect(function() down = false end)
end
drag(titleBar, main)

-- Mouse Fix: keep cursor usable in-game while GUI is open
-- ============================================================
local function updateMouseState()
    if main and main.Visible then
        pcall(function()
            UIS.MouseBehavior = Enum.MouseBehavior.Default
            UIS.MouseIconEnabled = true
        end)
    end
end

    RunSvc:BindToRenderStep("PetRageSuite_MouseFix", Enum.RenderPriority.Last.Value, updateMouseState)
end)

-- Backup: also run on Heartbeat in case the game locks there instead.
RunSvc.Heartbeat:Connect(updateMouseState)

local function newToggle(parent, opts)
    local row = make("TextButton", { Size = UDim2.new(1, 0, 0, 22), BackgroundTransparency = 1, Text = "", AutoButtonColor = false }, parent)
    local box = make("Frame", { Position = UDim2.fromOffset(0, 5), Size = UDim2.fromOffset(12, 12), BorderSizePixel = 0 }, row)
    local line = outline(box, col.boxLine)
    local fill = shade(box, col.fillA, col.fillB)
    local txt  = label(row, opts.name, UDim2.fromOffset(22, 0), UDim2.new(1, -22, 1, 0))
    local state = { value = opts.default and true or false }
    local function draw()
        local on = state.value
        fill.Enabled = on
        box.BackgroundColor3 = on and Color3.new(1, 1, 1) or col.box
        line.Color = on and Color3.fromRGB(160, 168, 182) or col.boxLine
        txt.TextColor3 = on and col.text or col.off
    end
    draw()
    row.MouseButton1Click:Connect(function()
        state.value = not state.value; draw()
        if opts.callback then pcall(opts.callback, state.value) end
    end)
    return state
end

local function newSlider(parent, opts)
    local min, max, step = opts.min or 0, opts.max or 1, opts.step or 0
    local holder = make("Frame", { Size = UDim2.new(1, 0, 0, 37), BackgroundTransparency = 1 }, parent)
    label(holder, opts.name, UDim2.new(), UDim2.new(1, 0, 0, 16), col.dim)
    local bar = make("TextButton", { Position = UDim2.fromOffset(0, 18), Size = UDim2.new(1, 0, 0, 15), BackgroundColor3 = col.box, BorderSizePixel = 0, Text = "", AutoButtonColor = false }, holder)
    outline(bar, col.boxLine)
    local fill = make("Frame", { BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0 }, bar)
    shade(fill, col.fillA, col.fillB)
    local val = label(bar, "", UDim2.new(), UDim2.fromScale(1, 1), col.text, Enum.TextXAlignment.Center)
    val.ZIndex = 3
    local function format(v)
        if opts.suffix then
            if step >= 1 then return string.format("%d%s", math.floor(v + 0.5), opts.suffix) end
            return string.format("%.2f%s", v, opts.suffix)
        end
        if step >= 1 then return string.format("%d", math.floor(v + 0.5)) end
        return string.format("%.2f", v)
    end
    local state = { value = opts.default or min }
    local function apply(v)
        if step > 0 then v = min + math.floor((v - min) / step + 0.5) * step end
        v = math.clamp(v, min, max)
        state.value = v
        fill.Size = UDim2.fromScale((v - min) / (max - min), 1)
        val.Text = format(v)
    end
    apply(state.value)
    local function fromX(x)
        local frac = (x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X
        apply(min + math.clamp(frac, 0, 1) * (max - min))
        if opts.callback then pcall(opts.callback, state.value) end
    end
    bar.InputBegan:Connect(function(i) if isPress(i) then sliding = fromX; fromX(i.Position.X) end end)
    return state
end

local function newDropdown(parent, opts)
    local dd = { _items = opts.items, _value = opts.default or opts.items[1] }
    local row = make("TextButton", { Size = UDim2.new(1, 0, 0, 22), BackgroundTransparency = 1, Text = "", AutoButtonColor = false }, parent)
    label(row, opts.name, UDim2.fromOffset(0, 0), UDim2.new(0.5, 0, 1, 0), col.dim)
    local valueLabel = label(row, tostring(dd._value), UDim2.new(0.5, 0, 0, 0), UDim2.new(0.5, -14, 1, 0), col.text, Enum.TextXAlignment.Right)
    label(row, "v", UDim2.new(1, -12, 0, 0), UDim2.fromOffset(12, 22), col.dim)
    local function refresh() valueLabel.Text = (dd._value == "" and "(None)") or tostring(dd._value) end
    refresh()
    row.MouseButton1Click:Connect(function()
        closeDropdown()
        local bp, bs = row.AbsolutePosition, row.AbsoluteSize
        local frame = make("Frame", { Position = UDim2.fromOffset(bp.X, bp.Y + bs.Y), Size = UDim2.fromOffset(bs.X, math.min(#dd._items * 20 + 4, 220)), BackgroundColor3 = col.box, BorderSizePixel = 0, ZIndex = 5 }, ddGui)
        outline(frame, col.boxLine)
        local scroll = make("ScrollingFrame", { Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, BorderSizePixel = 0, ScrollBarThickness = 4, CanvasSize = UDim2.new(0, 0, 0, #dd._items * 20) }, frame)
        make("UIListLayout", { SortOrder = Enum.SortOrder.LayoutOrder }, scroll)
        for _, opt in ipairs(dd._items) do
            local ib = make("TextButton", { Size = UDim2.new(1, 0, 0, 20), BackgroundTransparency = 1, Text = (opt == "" and "(None)") or tostring(opt), Font = font, TextSize = fsize, TextColor3 = (opt == dd._value) and col.accent or col.text, TextXAlignment = Enum.TextXAlignment.Left, AutoButtonColor = false }, scroll)
            make("UIPadding", { PaddingLeft = UDim.new(0, 6) }, ib)
            ib.MouseButton1Click:Connect(function() dd._value = opt; refresh(); closeDropdown(); if opts.callback then pcall(opts.callback, opt) end end)
            ib.MouseEnter:Connect(function() ib.TextColor3 = col.accent end)
            ib.MouseLeave:Connect(function() if opt ~= dd._value then ib.TextColor3 = col.text end end)
        end
        openDropdown = frame
    end)
    function dd:SetItems(newItems) self._items = newItems; if not table.find(newItems, self._value) then self._value = newItems[1]; refresh() end end
    return dd
end

UIS.InputBegan:Connect(function(i)
    if not openDropdown then return end
    if i.UserInputType ~= Enum.UserInputType.MouseButton1 and i.UserInputType ~= Enum.UserInputType.Touch then return end
    local m = UIS:GetMouseLocation()
    local p, s = openDropdown.AbsolutePosition, openDropdown.AbsoluteSize
    if m.X < p.X or m.X > p.X + s.X or m.Y < p.Y or m.Y > p.Y + s.Y then closeDropdown() end
end)

    row.MouseButton1Click:Connect(function() listening = true; valueLabel.Text = "Press..." end)

local function makeSection(scroll)
    local sec = {}
    function sec:toggle(o)  return newToggle(scroll, { name = o.name, default = o.default, callback = o.callback }) end
    function sec:slider(o)  return newSlider(scroll, { name = o.name, min = o.min, max = o.max, step = o.interval or 1, default = o.default, suffix = o.suffix, callback = o.callback }) end
    function sec:dropdown(o) return newDropdown(scroll, { name = o.name, items = o.items, default = o.default, callback = o.callback }) end
    function sec:button(o)  return newButton(scroll, { name = o.name, callback = o.callback }) end
    function sec:keybind(o) return newKeybind(scroll, { name = o.name, callback = o.callback }) end
    function sec:label(o)   return label(scroll, o.name or "", UDim2.new(), UDim2.new(1, 0, 0, 16), col.dim) end
    function sec:settings()  return makeSection(scroll) end
    return sec
end

local function makeColumn(tabContent)
    local c = {}
    function c:section(opts)
        local scroll = createPanel(tabContent, opts.name)
        return makeSection(scroll)
    end
    return c
end

local espAccum = 0
RunSvc.RenderStepped:Connect(function(dt)
    if not esp.enabled then
        for _, f in pairs(espFrames) do
            if f.frame.Visible or f.tracer.Visible then
                f.frame.Visible = false
                f.tracer.Visible = false
            end
        end
        return
    end
    espAccum = espAccum + dt
    if espAccum < 0.03 then return end
    espAccum = 0
    local myChar = LP.Character
    local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myHRP then return end
    local vp = cam.ViewportSize
    local ox, oy = vp.X / 2, vp.Y
    for _, plr in ipairs(Players:GetPlayers()) do
        local f = espFrames[plr] or createESPFrame()
        espFrames[plr] = f
        local char = plr.Character
        local hum  = char and char:FindFirstChildOfClass("Humanoid")
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")
        local head = char and char:FindFirstChild("Head")
        local visible = plr ~= LP and hrp and head and hum and hum.Health > 0
        if visible and esp.teamCheck and plr.Team ~= nil and plr.Team == LP.Team then visible = false end
        local dist = visible and (hrp.Position - myHRP.Position).Magnitude or 0
        if visible and dist > esp.maxDist then visible = false end
        if not visible then
            f.frame.Visible = false; f.tracer.Visible = false
        else
            local headPos, hOn = cam:WorldToViewportPoint(head.Position + Vector3.new(0, 1, 0))
            local footPos, fOn = cam:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))
            if not hOn or not fOn then
                f.frame.Visible = false; f.tracer.Visible = false
            else
                local top, bot = headPos.Y, footPos.Y
                local h = bot - top
                local w = h * 0.55
                local cx = (headPos.X + footPos.X) / 2
                f.frame.Position = UDim2.fromOffset(cx - w/2, top)
                f.frame.Size     = UDim2.fromOffset(w, h)
                f.frame.Visible  = esp.box
                f.stroke.Color   = esp.boxColor
                f.nameLabel.Visible    = esp.name
                f.nameLabel.Text       = (plr.DisplayName ~= plr.Name) and plr.DisplayName or plr.Name
                f.nameLabel.TextColor3 = esp.nameColor
                f.distLabel.Visible    = esp.dist
                f.distLabel.Text       = string.format("%d studs", math.floor(dist))
                f.healthBg.Visible     = esp.health
                local frac = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
                f.healthFill.Size = UDim2.new(1, 0, frac, 0)
                f.healthFill.Position = UDim2.new(0, 0, 1 - frac, 0)
                f.healthFill.BackgroundColor3 = esp.healthColor
                if esp.tracer then
                    f.tracer.Visible = true
                    local x1, y1 = ox, oy
                    local x2, y2 = cx, bot
                    local dx, dy = x2 - x1, y2 - y1
                    local len = math.sqrt(dx*dx + dy*dy)
                    f.tracer.Size = UDim2.fromOffset(len, 1)
                    f.tracer.AnchorPoint = Vector2.new(0.5, 0.5)
                    f.tracer.Position = UDim2.fromOffset((x1 + x2)/2, (y1 + y2)/2)
                    f.tracer.Rotation = math.deg(math.atan2(dy, dx))
                    f.tracer.BackgroundColor3 = esp.tracerColor
                else
                    f.tracer.Visible = false
                end
            end
        end
    end
    for plr, f in pairs(espFrames) do
        if not plr.Parent then f.frame:Destroy(); f.tracer:Destroy(); espFrames[plr] = nil end
    end
end)

local function getTarget(fov, partName, teamCheck, wallCheck)
    local closest, shortest = nil, fov
    local mouse = UIS:GetMouseLocation()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            local hum = p.Character:FindFirstChild("Humanoid")
            if hum and hum.Health > 0 then
                local skip = false
                if teamCheck and p.Team == LP.Team then skip = true end
                if not skip then
                    local part = p.Character:FindFirstChild(partName) or p.Character:FindFirstChild("HumanoidRootPart")
                    if part then
                        local screenPos, onScreen = cam:WorldToViewportPoint(part.Position)
                        if onScreen then
                            local dist = (Vector2.new(screenPos.X, screenPos.Y) - mouse).Magnitude
                            if dist < shortest then
                                if wallCheck then
                                    local ray = Ray.new(cam.CFrame.Position, part.Position - cam.CFrame.Position)
                                    local hit = workspace:FindPartOnRayWithIgnoreList(ray, {LP.Character, p.Character})
                                    if not hit then
                                        shortest = dist
                                        closest = { player = p, part = part, hum = hum }
                                    end
                                else
                                    shortest = dist
                                    closest = { player = p, part = part, hum = hum }
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    return closest
end

        local evasion_conn = nil
        M.start_evasion = function()
            if evasion_conn then evasion_conn:Disconnect() end
            local dt_buf = 0
            evasion_conn = run_service.Heartbeat:Connect(function(dt)
                if not config.evasion_enabled then return end
                local hrp = M.get_hrp(localplayer); local hum = localplayer.Character and localplayer.Character:FindFirstChildOfClass("Humanoid")
                if not hrp or not hum or hum.Health <= 0 then return end
                if hum:GetState() ~= Enum.HumanoidStateType.Physics then hum:ChangeState(Enum.HumanoidStateType.Physics) end
                dt_buf = dt_buf + dt; local interval = math.max((config.evasion_interval or 1) * 0.001, 0.001)
                if dt_buf >= interval then
                    dt_buf = 0; local iters = math.clamp(config.evasion_burst or 500, 100, 1000); local spd = config.evasion_spin_speed or 9999; local t = tick()
                    for i = 1, iters do
                        local frac = (i-1)/iters; local offset = compute_evasion_offset(config.evasion_mode or "Random", frac, 50000)
                        local dest = hrp.Position + offset
                        hrp.CFrame = CFrame.new(dest.X, dest.Y, dest.Z) * CFrame.Angles(math.rad(spd*t*1.0 % 360), math.rad(spd*t*0.73 % 360), math.rad(spd*t*0.41 % 360))
                    end
                    hrp.AssemblyLinearVelocity = Vector3.zero; hrp.AssemblyAngularVelocity = Vector3.zero
                end
            end)
        end
        M.stop_evasion = function() if evasion_conn then evasion_conn:Disconnect(); evasion_conn = nil end; config.evasion_enabled = false end

        local aa_conn_s, aa_orbit_angle = nil, 0

                if not hrp or not cam then return end
                local method = config.aa_method or "Static"; local t = tick() - aa_t0
                if method=="Pixelated" then
                    if hum then hum.AutoRotate=false end
                    local char=localplayer.Character; if char then for _,v in pairs(char:GetDescendants()) do if v:IsA("BasePart") then pcall(function() v.Size=v.Size*math.clamp((config.aa_pixel_scale or 1)/100,0.01,1) end) end end end
                    hrp.CFrame=hrp.CFrame*CFrame.Angles(math.rad(math.random(-360,360)),math.rad(math.random(-360,360)),math.rad(math.random(-360,360)))
                elseif method=="Static" then if hum then hum.AutoRotate=false end; hrp.CFrame=CFrame.new(hrp.Position)*CFrame.Angles(math.rad(180),0,0)
                elseif method=="Spin" then if hum then hum.AutoRotate=false end; hrp.CFrame=CFrame.new(hrp.Position)*CFrame.Angles(0,math.rad((config.aa_spin_speed or 9999)*t%360),0)
                elseif method=="Jitter" then if hum then hum.AutoRotate=false end; local tog=(math.floor(t*30))%2==0; hrp.CFrame=CFrame.new(hrp.Position)*CFrame.Angles(tog and math.rad(180) or 0,tog and math.rad(90) or math.rad(-90),0)
                elseif method=="Desync" then
                    if hum then hum.AutoRotate=false end
                    local rootPos=hrp.Position; local look=cam.CFrame.LookVector; local base=CFrame.lookAt(rootPos,rootPos+Vector3.new(look.X,0,look.Z))
                    hrp.CFrame=base*CFrame.Angles(math.rad(math.random(-360,360)),math.rad(math.random(-360,360)),math.rad(math.random(-360,360)))

                elseif method=="Sway" then if hum then hum.AutoRotate=false end; hrp.CFrame=CFrame.new(hrp.Position)*CFrame.Angles(math.cos(t*2)*math.rad(45),math.sin(t*3)*math.rad(90),math.sin(t*1.5)*math.rad(30))

                    local look=cam.CFrame.LookVector; local away=Vector3.new(-look.X,0,-look.Z); local rootPos=hrp.Position
                    if away.Magnitude>0.001 then hrp.CFrame=CFrame.lookAt(rootPos,rootPos+away)*CFrame.Angles(math.rad(15),math.rad(180),0) end
                elseif method=="Orbit" then
                    if hum then hum.AutoRotate=false end
                    aa_orbit_angle=aa_orbit_angle+dt*(config.aa_spin_speed or 9999)*0.05
                    local dirs={Vector3.new(math.cos(aa_orbit_angle),0,math.sin(aa_orbit_angle)),Vector3.new(0,math.sin(aa_orbit_angle),math.cos(aa_orbit_angle)),Vector3.new(math.cos(aa_orbit_angle*1.3),math.sin(aa_orbit_angle*0.7),math.sin(aa_orbit_angle*1.1))}

                    hrp.AssemblyLinearVelocity=Vector3.zero; hrp.AssemblyAngularVelocity=Vector3.zero
                elseif method=="Custom" then if hum then hum.AutoRotate=false end; hrp.CFrame=CFrame.new(hrp.Position)*CFrame.Angles(math.rad(config.aa_pitch or 0),math.rad(config.aa_yaw or 0),math.rad(config.aa_roll or 0))
                end
            end)
        end

        local function get_vs_order_offset(order, frac, i, ph, radius)
            local offset = Vector3.zero
            if order=="Triangle" then
                local seg=math.floor(frac*3); local f=(frac*3)-seg
                if seg==0 then offset=Vector3.new(0,radius,0):Lerp(Vector3.new(radius,-radius,0),f) elseif seg==1 then offset=Vector3.new(radius,-radius,0):Lerp(Vector3.new(-radius,-radius,0),f) else offset=Vector3.new(-radius,-radius,0):Lerp(Vector3.new(0,radius,0),f) end
            elseif order=="Circle" then offset=Vector3.new(math.cos(ph)*radius,0,math.sin(ph)*radius)
            elseif order=="Square" then
                local seg=math.floor(frac*4); local f=(frac*4)-seg
                if seg==0 then offset=Vector3.new(radius,-radius,0):Lerp(Vector3.new(radius,radius,0),f) elseif seg==1 then offset=Vector3.new(radius,radius,0):Lerp(Vector3.new(-radius,radius,0),f) elseif seg==2 then offset=Vector3.new(-radius,radius,0):Lerp(Vector3.new(-radius,-radius,0),f) else offset=Vector3.new(-radius,-radius,0):Lerp(Vector3.new(radius,-radius,0),f) end
            elseif order=="Pentagon" then local seg=math.floor(frac*5); local f=(frac*5)-seg; local a1=(seg/5)*math.pi*2; local a2=((seg+1)/5)*math.pi*2; offset=Vector3.new(math.cos(a1)*radius,0,math.sin(a1)*radius):Lerp(Vector3.new(math.cos(a2)*radius,0,math.sin(a2)*radius),f)
            elseif order=="Hexagon" then local seg=math.floor(frac*6); local f=(frac*6)-seg; local a1=(seg/6)*math.pi*2; local a2=((seg+1)/6)*math.pi*2; offset=Vector3.new(math.cos(a1)*radius,0,math.sin(a1)*radius):Lerp(Vector3.new(math.cos(a2)*radius,0,math.sin(a2)*radius),f)
            elseif order=="Octagon" then local seg=math.floor(frac*8); local f=(frac*8)-seg; local a1=(seg/8)*math.pi*2; local a2=((seg+1)/8)*math.pi*2; offset=Vector3.new(math.cos(a1)*radius,0,math.sin(a1)*radius):Lerp(Vector3.new(math.cos(a2)*radius,0,math.sin(a2)*radius),f)
            elseif order=="Star" then local seg=math.floor(frac*10); local f=(frac*10)-seg; local ro=radius; local ri=radius*0.4; local r1=seg%2==0 and ro or ri; local r2=seg%2==0 and ri or ro; local a1=(seg/10)*math.pi*2; local a2=((seg+1)/10)*math.pi*2; offset=Vector3.new(math.cos(a1)*r1,0,math.sin(a1)*r1):Lerp(Vector3.new(math.cos(a2)*r2,0,math.sin(a2)*r2),f)

            elseif order=="Cross" then local seg=math.floor(frac*4); local f=(frac*4)-seg; if seg==0 then offset=Vector3.new(-radius,0,0):Lerp(Vector3.new(radius,0,0),f) elseif seg==1 then offset=Vector3.new(radius,0,0):Lerp(Vector3.new(0,0,0),f) elseif seg==2 then offset=Vector3.new(0,0,-radius):Lerp(Vector3.new(0,0,radius),f) else offset=Vector3.new(0,0,radius):Lerp(Vector3.new(0,0,0),f) end
            elseif order=="Spiral" then local turns=3; local a=frac*math.pi*2*turns; local r=frac*radius; offset=Vector3.new(math.cos(a)*r,math.sin(frac*math.pi*2)*radius*0.5,math.sin(a)*r)
            elseif order=="Zigzag" then local segs=20; local seg=math.floor(frac*segs); local f=(frac*segs)-seg; local x1=(seg%2==0) and -radius or radius; local x2=(seg%2==0) and radius or -radius; local z1=(seg/segs)*radius*2-radius; local z2=((seg+1)/segs)*radius*2-radius; offset=Vector3.new(x1,0,z1):Lerp(Vector3.new(x2,0,z2),f)
            elseif order=="Wave" then local x=(frac-0.5)*radius*2; offset=Vector3.new(x,math.sin(frac*math.pi*6)*radius,0)
            elseif order=="Figure8" then offset=Vector3.new(math.sin(ph)*radius,math.sin(ph*2)*radius*0.25,math.sin(ph)*math.cos(ph)*radius)
            elseif order=="Lemniscate" then local cos2=math.cos(2*ph); local rr=cos2>=0 and math.sqrt(cos2)*radius or 0; offset=Vector3.new(math.cos(ph)*rr,0,math.sin(ph)*rr)
            elseif order=="Rose" then local a=frac*math.pi; local rr=math.cos(3*a)*radius; offset=Vector3.new(rr*math.cos(a),0,rr*math.sin(a))
            elseif order=="Butterfly" then local rr=(math.exp(math.cos(ph))-2*math.cos(4*ph))*radius*0.3; offset=Vector3.new(math.sin(ph)*rr,math.cos(ph*2)*radius*0.25,math.cos(ph)*rr)

        local function get_rage_offset(strat, t_hrp, base_pos)
            local look = t_hrp.CFrame.LookVector; if look.Magnitude < 0.001 then look = Vector3.new(0, 0, -1) end
            local offset = Vector3.zero; local wild = CFrame.Angles(0, 0, 0)
            if strat == "Orbit" then
                local spin_rad = tick() * (config.rage_orbit_speed or 500) / 100; local radius = math.max(config.rage_tp_radius or 5, 1); local height = config.rage_tp_height or -4
                offset = Vector3.new(math.cos(spin_rad)*radius, height, math.sin(spin_rad)*radius)
                wild = CFrame.lookAt(Vector3.zero, Vector3.new(math.cos(spin_rad+math.pi),0,math.sin(spin_rad+math.pi)))
            else
                local pos_mode = string.lower(config.rage_position or "inside"); local pos_dist = config.rage_tp_radius or 2
                if pos_mode == "front" then offset = look*pos_dist elseif pos_mode == "behind" then offset = -look*pos_dist elseif pos_mode == "above" then offset = Vector3.new(0, math.abs(config.rage_tp_height or 15)+5, 0) elseif pos_mode == "below" then offset = Vector3.new(0, -(math.abs(config.rage_tp_height or 4)+2), 0) end
                local mode = config.rage_type or "Melee"
                if mode == "Gun" then offset = Vector3.new(0, math.max(config.rage_tp_height or 15, 5), 0) elseif mode == "Knife" then offset = -look*(config.knife_dist or 3) end
                if config.rage_spin then local t = tick(); wild = CFrame.Angles(math.rad(1e70*t*1.0 % 360),math.rad(1e70*t*0.73 % 360),math.rad(1e70*t*0.41 % 360)) end
            end
            return offset, wild
        end

            local function computeVoidDriftDir(t)
                local nx, nz, amplitude, frequency = 0, 0, 1, 0.0001
                for _ = 1, 4 do
                    nx = nx + math.noise(t * frequency, 0) * amplitude
                    nz = nz + math.noise(0, t * frequency) * amplitude
                    frequency = frequency * 2.37
                    amplitude = amplitude * 0.5
                end
                nx = nx + math.sin(t * 0.00213) * math.cos(t * 0.00344) * 0.2
                nz = nz + math.cos(t * 0.00131) * math.sin(t * 0.00579) * 0.2
                local len = math.sqrt(nx * nx + nz * nz)
                if len < 0.001 then return math.cos(t * 0.1), math.sin(t * 0.1) end
                return nx / len, nz / len
            end

                elseif VM.mode == "CAMERA" and workspace.CurrentCamera then
                    offset = workspace.CurrentCamera.CFrame.LookVector * VM.distance
                elseif VM.mode == "DIRECTIONAL" then
                    offset = vmDirLimitOffset()
                else
                    local r1 = VM.distance * (0.60 + 0.40 * math.sin(phase * 2.3))
                    local r2 = VM.distance * (0.25 + 0.15 * math.sin(phase * 5.7))
                    local r3 = VM.distance * (0.10 + 0.10 * math.sin(phase * 11.3))

                    local oZ = math.sin(phase * 7.1) * r1 + math.sin(phase * 13.4) * r2 + math.sin(phase * 21.9) * r3
                    local oY = math.sin(phase * 9) * r1 * 0.4 + math.sin(phase * 17) * r2 * 0.3

                    local jY = math.noise(0, phase * 6, 0) * VM.jitter * 1.5 + (math.random() - 0.5) * VM.jitter * 1.2
                    local jZ = math.noise(0, 0, phase * 6) * VM.jitter * 3 + (math.random() - 0.5) * VM.jitter * 2.5
                    offset = Vector3.new(oX + jX, oY + jY, oZ + jZ)

do
    local colO = makeColumn(tabs[9].content)
    xpcall(function()
        local lp = LP
        local cfg = { speed = 500, distance = 5, height = -4, maxDistance = 800, useDesync = false }
        local S = { on = false, conn = nil, charConn = nil, diedConn = nil, lastAngle = 0, lastPos = nil,
            targetReady = false, lobbyLeft = 0, cooldown = 0, client = nil, renderBound = false }
        local RENDER_NAME = "SniflvaOrbitSimpleDesync"
        local function getRoot(char) return char and (char:FindFirstChild("HumanoidRootPart") or char.PrimaryPart) end
        local function alive(char) local h = char and char:FindFirstChildOfClass("Humanoid"); return h ~= nil and h.Health > 0 end
        local function inLobby()
            local ok, res = pcall(function()
                local gui = lp:FindFirstChild("PlayerGui"); if not gui then return false end
                local main = gui:FindFirstChild("MainGui"); if not main then return false end
                local frame = main:FindFirstChild("MainFrame"); if not frame then return false end
                local lobby = frame:FindFirstChild("Lobby"); if not lobby then return false end
                local currency = lobby:FindFirstChild("Currency"); if not currency then return false end
                return currency.Visible
            end)
            return ok and res == true
        end
        local function resetState() S.client = nil; S.targetReady = false; S.lobbyLeft = 0 end
        local function stopRender() if S.renderBound then pcall(function() RunSvc:UnbindFromRenderStep(RENDER_NAME) end); S.renderBound = false end end
        local function startRender()
            stopRender(); S.renderBound = true
            RunSvc:BindToRenderStep(RENDER_NAME, Enum.RenderPriority.First.Value, function()
                if not (cfg.useDesync and S.on) then return end
                if S.cooldown > tick() then return end
                local char = lp.Character; if not alive(char) then return end
                local root = getRoot(char); if root and S.client then root.CFrame = S.client end
            end)
        end
        local function nearest()
            local myRoot = getRoot(lp.Character); if not myRoot then return nil end
            local best, bestD = nil, cfg.maxDistance
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= lp then
                    local c = plr.Character; local r = getRoot(c)
                    if r and alive(c) then local d = (r.Position - myRoot.Position).Magnitude; if d < bestD then bestD = d; best = plr end end
                end
            end
            return best
        end
        local function hookChar(char)
            S.lastPos = nil
            if S.diedConn then pcall(function() S.diedConn:Disconnect() end); S.diedConn = nil end
            local hum = char and char:WaitForChild("Humanoid", 5)
            if not hum or not S.on then return end
            if cfg.useDesync then startRender() end
            S.diedConn = hum.Died:Connect(function() resetState() end)
        end
        local function step(dt)
            local char = lp.Character; local root = getRoot(char); if not root then return end
            if not S.lastPos then S.lastPos = root.Position
            else
                local d = (root.Position - S.lastPos).Magnitude; S.lastPos = root.Position
                if d > 80 then S.cooldown = tick() + 1.2; resetState(); return end
            end
            if S.cooldown > tick() then return end
            if inLobby() then resetState(); return end
            if not S.targetReady then
                if S.lobbyLeft == 0 then S.lobbyLeft = tick(); return end
                if tick() - S.lobbyLeft >= 1 then S.targetReady = true else return end
            end
            if not alive(char) then return end
            if cfg.useDesync then S.client = root.CFrame end
            local target = nearest()
            if target and target.Character and alive(target.Character) then
                local tr = getRoot(target.Character)
                if tr then
                    S.lastAngle = S.lastAngle + dt * cfg.speed * math.rad(360)
                    local radius = cfg.distance + math.random(-3, 3); local height = cfg.height + math.random(-6, 6)
                    local angle = S.lastAngle + math.rad(math.random(-20, 20)); local center = tr.Position
                    root.CFrame = CFrame.new(center + Vector3.new(math.cos(angle) * radius, height, math.sin(angle) * radius), center)
                end
            end
        end
        local function setOrbit(on)
            if S.conn then pcall(function() S.conn:Disconnect() end); S.conn = nil end
            if S.charConn then pcall(function() S.charConn:Disconnect() end); S.charConn = nil end
            if S.diedConn then pcall(function() S.diedConn:Disconnect() end); S.diedConn = nil end
            stopRender(); resetState()
            S.on = on and true or false

            if lp.Character then task.spawn(hookChar, lp.Character) end
        end

        sec1:slider({ name = "Speed", min = 100, max = 1000, interval = 1, default = 500, callback = function(v) cfg.speed = v end })
        sec1:slider({ name = "Distance", min = 3, max = 15, interval = 0.1, default = 5, callback = function(v) cfg.distance = v end })

        local orbitAngle, orbitElapsed, orbitCurrentRadius = 0, 0, 8
        local orbitConn = nil

            local root = getFeatureRoot(); if not root then return nil end

                    if hrp and hum and hum.Health > 0 then local d = (root.Position - hrp.Position).Magnitude; if d < bestDist then best, bestDist = plr, d end end
                end
            end
            return best
        end
        local function startOrbitModes()
            if orbitConn then orbitConn:Disconnect(); orbitConn = nil end
            orbitAngle, orbitElapsed = 0, 0; orbitCurrentRadius = orbitCfg.radius

                local targetPos = hrp.Position

                orbitAngle = orbitAngle + math.rad(orbitCfg.speed) * dt; orbitElapsed = orbitElapsed + dt
                local r, h = orbitCfg.radius, orbitCfg.height; local x, y, z = 0, h, 0
                if orbitCfg.mode == 'Circle' then x, z = math.cos(orbitAngle)*r, math.sin(orbitAngle)*r
                elseif orbitCfg.mode == 'Figure 8' then x, z = math.sin(orbitAngle)*r, math.sin(orbitAngle)*math.cos(orbitAngle)*r*0.7
                elseif orbitCfg.mode == 'Spiral In' then orbitCurrentRadius = math.max(1, orbitCurrentRadius - dt*2); x, z = math.cos(orbitAngle)*orbitCurrentRadius, math.sin(orbitAngle)*orbitCurrentRadius
                elseif orbitCfg.mode == 'Spiral Out' then orbitCurrentRadius = math.min(r*2, orbitCurrentRadius + dt*2); x, z = math.cos(orbitAngle)*orbitCurrentRadius, math.sin(orbitAngle)*orbitCurrentRadius
                elseif orbitCfg.mode == 'Bounce' then x, z = math.cos(orbitAngle)*r, math.sin(orbitAngle)*r; y = h + math.sin(orbitElapsed*3)*5 end
                local dst = targetPos + Vector3.new(x,y,z); local smooth = root.Position:Lerp(dst, math.clamp(orbitCfg.smoothing, 0.01, 1))
                if orbitCfg.faceTarget then root.CFrame = CFrame.new(smooth, Vector3.new(targetPos.X, smooth.Y, targetPos.Z)) else root.CFrame = CFrame.new(smooth) end
            end)
