local function getClosest()
    if not root then return nil end
    local best, bestDist = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LP or not p.Character then continue end
        local hrp = p.Character:FindFirstChild("HumanoidRootPart")
        if not hrp then continue end
        local d = (root.Position - hrp.Position).Magnitude
        if d < bestDist then bestDist = d; best = p end
    end
    return best
end

local function getTarget(mode)
    if mode == "Closest" then return getClosest() end
    local list = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            table.insert(list, p)
        end
    end
    if #list == 0 then return nil end
    if mode == "Random" then return list[math.random(1, #list)]
    elseif mode == "Weakest" then
        table.sort(list, function(a, b)
            local ha = a.Character:FindFirstChildOfClass("Humanoid")
            local hb = b.Character:FindFirstChildOfClass("Humanoid")
            return (ha and ha.Health or 0) < (hb and hb.Health or 0)
        end)
        return list[1]
    elseif mode == "Strongest" then
        table.sort(list, function(a, b)
            local ha = a.Character:FindFirstChildOfClass("Humanoid")
            local hb = b.Character:FindFirstChildOfClass("Humanoid")
            return (ha and ha.Health or 0) > (hb and hb.Health or 0)
        end)
        return list[1]
    end
    return list[1]
end

local function predictPos(hrp, lead)
    if not hrp then return Vector3.zero end
    return hrp.Position + hrp.AssemblyLinearVelocity * lead
end

local function driftDir3D(t)
    local nx, ny, nz = 0, 0, 0
    local amp, freq = 1, 0.0001
    for _ = 1, 4 do
        nx = nx + math.noise(t * freq, 0, 0) * amp
        ny = ny + math.noise(0, t * freq, 0) * amp
        nz = nz + math.noise(0, 0, t * freq) * amp
        freq = freq * 2.37; amp = amp * 0.5
    end
    local len = math.sqrt(nx * nx + ny * ny + nz * nz)
    if len < 0.001 then return 1, 0, 0 end
    return nx / len, ny / len, nz / len
end

local function safeFilter(pos)
    if cfg.safeRadius > 0 then
        for _, p in ipairs(Players:GetPlayers()) do
            if p == LP or not p.Character then continue end
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if not hrp then continue end
            local d = pos - hrp.Position
            if d.Magnitude < cfg.safeRadius then
                local dir = d.Magnitude > 0.01 and -d.Unit or Vector3.new(1, 0, 0)
                pos = hrp.Position + dir * cfg.safeRadius
            end
        end
    end
    return pos
end

local function getScreenPoint()
    if not Camera then
        return Vector2.new(math.random(), math.random())
    end
    local vp = Camera.ViewportSize
    local mode = cfg.tpPlacement
    local x, y

    if mode == "Corners" then
        local corners = { Vector2.new(0, 0), Vector2.new(vp.X, 0), Vector2.new(0, vp.Y), Vector2.new(vp.X, vp.Y) }
        local c = corners[math.random(1, 4)]
        x = c.X; y = c.Y
    elseif mode == "Edges" then
        local edge = math.random(1, 4)
        if edge == 1 then x = math.random() * vp.X; y = 0
        elseif edge == 2 then x = math.random() * vp.X; y = vp.Y
        elseif edge == 3 then x = 0; y = math.random() * vp.Y
        else x = vp.X; y = math.random() * vp.Y end
    elseif mode == "Quadrants" then
        local q = math.random(1, 4)
        local hw, hh = vp.X / 2, vp.Y / 2
        if q == 1 then x = math.random() * hw; y = math.random() * hh
        elseif q == 2 then x = hw + math.random() * hw; y = math.random() * hh
        elseif q == 3 then x = math.random() * hw; y = hh + math.random() * hh
        else x = hw + math.random() * hw; y = hh + math.random() * hh end
    elseif mode == "Weighted" then
        local roll = math.random()
        if roll < cfg.tpCornerWeight then
            local corners = { Vector2.new(0, 0), Vector2.new(vp.X, 0), Vector2.new(0, vp.Y), Vector2.new(vp.X, vp.Y) }
            local c = corners[math.random(1, 4)]
            x = c.X; y = c.Y
        elseif roll < cfg.tpCornerWeight + cfg.tpEdgeWeight then
            local edge = math.random(1, 4)
            if edge == 1 then x = math.random() * vp.X; y = 0
            elseif edge == 2 then x = math.random() * vp.X; y = vp.Y
            elseif edge == 3 then x = 0; y = math.random() * vp.Y
            else x = vp.X; y = math.random() * vp.Y end
        else
            x = math.random() * vp.X
            y = math.random() * vp.Y
        end
    else
        x = math.random() * vp.X
        y = math.random() * vp.Y
        if not cfg.tpFullScreen then
            x = vp.X * 0.25 + math.random() * vp.X * 0.5
            y = vp.Y * 0.25 + math.random() * vp.Y * 0.5
        end
    end

    if cfg.tpScreenJitter > 0 then
        x = x + (math.random() - 0.5) * vp.X * cfg.tpScreenJitter
        y = y + (math.random() - 0.5) * vp.Y * cfg.tpScreenJitter
    end
    if cfg.tpClampScreen then
        x = math.clamp(x, 0, vp.X)
        y = math.clamp(y, 0, vp.Y)
    end

    return Vector2.new(x, y)
end

local function getTpEverywherePosition()
    local dist
    if cfg.tpRandDist then
        dist = cfg.tpDistMin + math.random() * (cfg.tpDistMax - cfg.tpDistMin)
    else
        dist = cfg.tpDistMin
    end
    if cfg.tpNegBias and math.random() < 0.5 then dist = -math.abs(dist) end

    if not Camera or not cfg.tpViewportRay then
        local rx = (math.random() * 2 - 1) * cfg.xRange
        local ry = cfg.yMin + math.random() * (cfg.yMax - cfg.yMin)
        local rz = (math.random() * 2 - 1) * cfg.zRange
        return Vector3.new(cfg.baseX + rx, ry, cfg.baseZ + rz)
    end

    local sp = getScreenPoint()
    local ray = Camera:ViewportPointToRay(sp.X, sp.Y, math.abs(dist))
    local base = ray.Origin

    local vertMult = cfg.tpVertMin + math.random() * (cfg.tpVertMax - cfg.tpVertMin)
    local vert = cfg.tpVertSpike and (math.random() * 2 - 1) * math.abs(dist) * vertMult or 0
    if cfg.tpNegBias then vert = vert - math.abs(vert) * 0.3 end
    if cfg.preserveY then vert = 0 end

    local y = cfg.preserveY and (root and root.Position.Y or 0) or (base.Y + vert)
    return Vector3.new(base.X, y, base.Z)
end

local function stepVoid()
    local m = cfg.voidMode
    if m == "TP Everywhere" then
        return getTpEverywherePosition()
    elseif m == "Chaos" then
        return Vector3.new(
            cfg.baseX + (math.random() * 2 - 1) * cfg.xRange,
            cfg.yMin + math.random() * (cfg.yMax - cfg.yMin),
            cfg.baseZ + (math.random() * 2 - 1) * cfg.zRange
        )
    elseif m == "Drift" then
        local dx, dy, dz = driftDir3D(state.elapsed)
        local target = Vector3.new(dx, dy, dz).Unit
        state.driftDir = state.driftDir:Lerp(target, math.clamp(cfg.driftChaos * (1/60) * 10, 0, 1))
        state.driftPos = (root and root.Position or Vector3.zero) + state.driftDir * cfg.driftSpeed * (1/60)
        return state.driftPos
    elseif m == "Circle" then
        state.circleAng = state.circleAng + math.rad(cfg.circleSpeed) * (1/60)
        return Vector3.new(
            cfg.baseX + math.cos(state.circleAng) * cfg.circleRadius,
            cfg.baseY + math.sin(state.circleAng * 1.3) * 50,
            cfg.baseZ + math.sin(state.circleAng) * cfg.circleRadius
        )
    elseif m == "Spiral" then
        state.spiralR = math.min(cfg.spiralRadiusMax, state.spiralR + cfg.spiralRate * (1/60))
        state.circleAng = state.circleAng + math.rad(60) * (1/60)
        return Vector3.new(
            cfg.baseX + math.cos(state.circleAng) * state.spiralR,
            cfg.baseY + math.sin(state.circleAng * 0.5) * state.spiralR * 0.3,
            cfg.baseZ + math.sin(state.circleAng) * state.spiralR
        )
    elseif m == "OrbitTarget" then
        local t = getClosest()
        if t and t.Character then
            local hrp = t.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                local ang = state.elapsed * 6
                return hrp.Position + Vector3.new(math.cos(ang) * 5, 2, math.sin(ang) * 5)
            end
        end
        return getTpEverywherePosition()
    elseif m == "Predictive" then
        local t = getClosest()
        if t and t.Character then
            local hrp = t.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                return predictPos(hrp, cfg.predictiveLead) + Vector3.new(
                    (math.random() - 0.5) * 100,
                    (math.random() - 0.5) * 100,
                    (math.random() - 0.5) * 100
                )
            end
        end
        return getTpEverywherePosition()
    end
    return root and root.Position or Vector3.zero
end

function startOrbit()
    killConn("orbit")
    if hum then pcall(function() hum:ChangeState(Enum.HumanoidStateType.Physics) end) end

    local angle = 0
    local localElapsed = 0
    local currentRadius = orbitCfg.dist
    local snapTimer = 0
    local pulseT = 0

    conns.orbit = RunService.Heartbeat:Connect(function(dt)
        if not orbitCfg.enabled then killConn("orbit"); return end
        if not root or not root.Parent then return end

        if hum then pcall(function()
            if hum:GetState() ~= Enum.HumanoidStateType.Physics then
                hum:ChangeState(Enum.HumanoidStateType.Physics)
            end
        end) end

        local target = getTarget(orbitCfg.targetMode)
        local targetPos = root.Position
        if target and target.Character then
            local hrp = target.Character:FindFirstChild("HumanoidRootPart")
            if hrp and (root.Position - hrp.Position).Magnitude <= orbitCfg.lockDist then
                targetPos = hrp.Position
                if orbitCfg.predict then
                    targetPos = targetPos + hrp.AssemblyLinearVelocity * orbitCfg.predictStr
                end
            end
        end

        local speed = orbitCfg.reverse and -orbitCfg.speed or orbitCfg.speed
        if orbitCfg.speedRandom then
            speed = orbitCfg.speedRandMin + math.random() * (orbitCfg.speedRandMax - orbitCfg.speedRandMin)
            if orbitCfg.reverse then speed = -speed end
        end

        angle = (angle + math.rad(speed) * dt) % (2 * math.pi)
        localElapsed = localElapsed + dt
        pulseT = pulseT + dt

        local r = orbitCfg.randRad and math.random(orbitCfg.randRadMin, orbitCfg.randRadMax) or orbitCfg.dist
        if orbitCfg.distPulse then
            r = r + math.sin(pulseT * orbitCfg.distPulseSpeed * 2 * math.pi) * orbitCfg.distPulseAmp
        end

        local h = orbitCfg.height
        if orbitCfg.heightOsc then
            h = h + math.sin(localElapsed * orbitCfg.heightOscSpeed * 2 * math.pi) * orbitCfg.heightOscAmp
        end

        local x, y, z = 0, 0, 0
        local om = orbitCfg.mode
        local ax = orbitCfg.axis

        if om == "Circle" then
            if ax == "XZ" then x = math.cos(angle)*r; z = math.sin(angle)*r; y = h
            elseif ax == "XY" then x = math.cos(angle)*r; y = math.sin(angle)*r; z = h
            elseif ax == "YZ" then y = math.cos(angle)*r; z = math.sin(angle)*r; x = h end
        elseif om == "Figure 8" then
            x = math.sin(angle)*r; z = math.sin(angle)*math.cos(angle)*r*0.7; y = h
        elseif om == "Spiral In" then
            currentRadius = math.max(1, currentRadius - dt * orbitCfg.dist * 0.2)
            x = math.cos(angle)*currentRadius; z = math.sin(angle)*currentRadius; y = h
        elseif om == "Spiral Out" then
            currentRadius = math.min(orbitCfg.dist*2, currentRadius + dt * orbitCfg.dist * 0.2)
            x = math.cos(angle)*currentRadius; z = math.sin(angle)*currentRadius; y = h
        elseif om == "Lemniscate" then
            local denom = 1 + math.sin(angle)^2
            x = r*math.cos(angle)/denom
            z = r*math.sin(angle)*math.cos(angle)/denom
            y = h
        end

        if orbitCfg.movementEnabled then
            if orbitCfg.moveMode == "Camera" and Camera then
                x = x + Camera.CFrame.RightVector.X * orbitCfg.moveRight + Camera.CFrame.LookVector.X * orbitCfg.moveForward
                y = y + orbitCfg.moveUp
                z = z + Camera.CFrame.RightVector.Z * orbitCfg.moveRight + Camera.CFrame.LookVector.Z * orbitCfg.moveForward
            else
                x = x + orbitCfg.moveRight
                y = y + orbitCfg.moveUp
                z = z + orbitCfg.moveForward
            end
        end

        x = x + orbitCfg.offX
        y = y + orbitCfg.offY
        z = z + orbitCfg.offZ
        local dst = targetPos + Vector3.new(x, y, z)

        if orbitCfg.snapBack then
            snapTimer = snapTimer + dt
            if snapTimer >= orbitCfg.snapInterval then
                snapTimer = 0
                dst = targetPos + Vector3.new(orbitCfg.offX, h + orbitCfg.offY, orbitCfg.offZ)
            end
        end

        local alpha = math.min(1, 1 - (1 - orbitCfg.lerp)^(dt * 60))
        local smooth = root.Position:Lerp(dst, alpha)

        pcall(function()
            if orbitCfg.faceTarget then
                local cf = CFrame.new(smooth, Vector3.new(targetPos.X, smooth.Y, targetPos.Z))
                cf = cf * CFrame.Angles(math.rad(orbitCfg.pitchOff), math.rad(orbitCfg.yawOff), 0)
                root.CFrame = cf
            else
                root.CFrame = CFrame.new(smooth)
            end
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
        end)
    end)
end

local function buildThreatMap(myPos)
    local map = {}
    if not myPos then return map end
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LP or not p.Character then continue end
        local hrp = p.Character:FindFirstChild("HumanoidRootPart")
        if not hrp then continue end
        local head = p.Character:FindFirstChild("Head") or hrp
        local look = head.CFrame.LookVector
        local eye = head.Position
        local toMe = myPos - eye
        local dist = toMe.Magnitude
        if dist < 0.01 then continue end
        local aimDot = look:Dot(toMe.Unit)
        local vel = hrp.AssemblyLinearVelocity
        local velDot = vel.Magnitude > 1 and vel.Unit:Dot(toMe.Unit) or 0
        local score = math.max(0, aimDot) * 45 + math.max(0, velDot) * 20
        if p.Character:FindFirstChildOfClass("Tool") then score = score + 25 end
        score = score + (predCfg.infiniteRange and 30 or math.max(0, 1 - dist / 250) * 30)
        map[p] = { hrp = hrp, dist = dist, aimDot = aimDot, velDot = velDot, score = score }
    end
    return map
end

local function perpDodgeDir(fromPos, myPos)
    local flat = Vector3.new(fromPos.X - myPos.X, 0, fromPos.Z - myPos.Z)
    if flat.Magnitude < 0.01 then flat = Vector3.new(1, 0, 0) end
    flat = flat.Unit
    local p1 = Vector3.new(-flat.Z, 0, flat.X)
    local p2 = Vector3.new(flat.Z, 0, -flat.X)
    return math.random(0, 1) == 0 and p1 or p2
end

local function raycastLineOfFire(enemyHrp, myPos)
    if not enemyHrp then return false, 0 end
    local origin = enemyHrp.Position + Vector3.new(0, 1.5, 0)
    local target = myPos + Vector3.new(0, 1.5, 0)
    local dir = target - origin
    local dist = dir.Magnitude
    if dist < 0.01 then return false, 0 end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { LP.Character, enemyHrp.Parent }
    local result = workspace:Raycast(origin, dir.Unit * dist, params)
    return result == nil, dist
end

local function getAimOrigin(plr)
    if not plr or not plr.Character then return nil end
    return plr.Character:FindFirstChild("Head")
        or plr.Character:FindFirstChild("CameraBone", true)
        or plr.Character:FindFirstChild("HumanoidRootPart")
end

local function predictEnemyAim(plr, lead, myHrp)
    if not plr or not plr.Character or not myHrp then return nil end
    local origin = getAimOrigin(plr)
    if not origin then return nil end
    local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local eVel = hrp.AssemblyLinearVelocity
    local mVel = myHrp.AssemblyLinearVelocity
    local ePos = origin.Position + eVel * lead
    local mPos = myHrp.Position + mVel * lead
    local dir = mPos - ePos
    if dir.Magnitude < 0.01 then return nil end
    local futureDot = origin.CFrame.LookVector:Dot(dir.Unit)
    return { futureDot = futureDot, dist = dir.Magnitude }
end

local function animationThreat(plr)
    if not predCfg.animDetect then return false end
    local c = plr.Character
    if not c then return false end
    local h = c:FindFirstChildOfClass("Humanoid")
    if not h then return false end
    local animator = h:FindFirstChildOfClass("Animator")
    if not animator then return false end
    for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
        local name = (track.Animation and track.Animation.Name or ""):lower()
        if name:find("shoot") or name:find("fire") or name:find("attack")
            or name:find("swing") or name:find("slash") or name:find("punch")
            or name:find("cast") or name:find("thrust") then
            return true
        end
    end
    return false
end

function startPrediction()
    killConn("pred")
    conns.pred = RunService.Heartbeat:Connect(function(dt)
        if not predCfg.enabled then killConn("pred"); return end
        if not root or not root.Parent then return end
        if hum and hum.Health <= 0 then return end

        local myPos = root.Position
        local now = tick()
        local map = buildThreatMap(myPos)
        local primary, topScore = nil, 0
        for p, d in pairs(map) do
            if d.score > topScore then topScore = d.score; primary = { p = p, d = d } end
        end
        if not primary then return end

        local effectiveRadius = predCfg.infiniteRange and 5e12 or predCfg.dangerRadius
        if primary.d.dist > effectiveRadius then return end

        local cooldown = predCfg.cooldown
        if predCfg.reactionLadder then
            if now - predState.lastShot < 1.5 then
                predState.rungIdx = math.min(predState.rungIdx + 1, 5)
            else
                predState.rungIdx = 0
            end
            cooldown = cooldown / (1 + predState.rungIdx * 0.5)
        end
        if now - predState.lastDodgeTick < cooldown then return end

        local threat = false
        local fired = predState.shootCD[primary.p] and (now - predState.shootCD[primary.p]) < predCfg.reactTime
        if fired then threat = true end
        if primary.d.aimDot > predCfg.shootDot then threat = true end
        if primary.d.velDot > 0.7 and primary.d.hrp.AssemblyLinearVelocity.Magnitude > predCfg.speedThreshold then threat = true end

        if predCfg.raycast then
            local clear = raycastLineOfFire(primary.d.hrp, myPos)
            if clear and primary.d.aimDot > 0.6 then threat = true end
        end

        if predCfg.predictAim and not threat then
            local prediction = predictEnemyAim(primary.p, predCfg.predictAimLead, root)
            if prediction and prediction.futureDot > 0.85 and prediction.dist < effectiveRadius then
                threat = true
            end
        end

        if predCfg.animDetect and not threat then
            if animationThreat(primary.p) then threat = true end
        end

        if not threat then return end

        predState.lastDodgeTick = now
        predState.lastShot = now
        local dir = perpDodgeDir(primary.d.hrp.Position, myPos)
        local newPos = myPos + dir * predCfg.dodgeDist

        if predCfg.multiPoint then
            local alt = Vector3.new(-dir.Z, 0, dir.X)
            if math.random() < 0.5 then alt = -alt end
            newPos = myPos + dir * (predCfg.dodgeDist * 0.7) + alt * (predCfg.dodgeDist * 0.5)
        end
        if predCfg.backstep then
            local toward = primary.d.hrp.Position - myPos
            if toward.Magnitude > 0.01 then
                newPos = newPos - toward.Unit * (predCfg.dodgeDist * 0.3)
            end
        end
        if predCfg.verticalSpike then
            newPos = newPos + Vector3.new(0, math.random() * 15, 0)
        end

        for i = 1, 3 do
            pcall(function()
                root.CFrame = CFrame.new(newPos + Vector3.new((math.random() - 0.5) * 4, 0, (math.random() - 0.5) * 4))
                root.AssemblyLinearVelocity = dir * 150
            end)
            RunService.Heartbeat:Wait()
        end

        if predCfg.notifyAnywhere or primary.d.dist < 500 then
            if predCfg.notify then
                Notify("pred: " .. primary.p.Name .. " [" .. math.floor(primary.d.dist) .. "]", 1.5)
            end
        end
    end)
end

function stopPrediction()
    killConn("pred")
    predCfg.enabled = false
end

function startAntiAim()
    killConn("antiAim")
    conns.antiAim = RunService.Heartbeat:Connect(function(dt)
        if not aaCfg.enabled then killConn("antiAim"); return end
        if not root then return end

        aaState.microTimer = aaState.microTimer + dt
        aaState.breathAcc = aaState.breathAcc + dt
        aaState.rollAcc = aaState.rollAcc + dt
        aaState.upDownT = aaState.upDownT + dt

        local spd = aaCfg.speed
        if aaCfg.randomSpeed then
            aaState.randTimer = aaState.randTimer + dt
            if aaState.randTimer > 0.1 then
                aaState.randTimer = 0
                aaState.randMul = 0.2 + math.random() * 0.8
            end
            spd = spd * aaState.randMul
        end

        local extraYaw = 0
        local extraPitch = 0
        if aaCfg.fakeAngle then
            extraYaw = extraYaw + aaCfg.fakeYaw
            extraPitch = extraPitch + aaCfg.fakePitch
        end
        if aaCfg.breathe then
            extraPitch = extraPitch + math.sin(aaState.breathAcc * aaCfg.breatheSpeed * 2 * math.pi) * aaCfg.breatheAmp
        end
        if aaCfg.microJitter and aaState.microTimer >= (1 / math.max(aaCfg.microJitterSpeed, 1)) then
            aaState.microTimer = 0
            extraYaw = extraYaw + (math.random() - 0.5) * aaCfg.microJitterAmp * 2
            extraPitch = extraPitch + (math.random() - 0.5) * aaCfg.microJitterAmp
        end
        if aaCfg.desyncMode then extraYaw = extraYaw + aaCfg.desyncOffset end
        if aaCfg.randomFlip and math.random() < aaCfg.flipChance then extraYaw = extraYaw + 180 end

        local rollAngle = 0
        if aaCfg.rollEnabled then
            rollAngle = (aaState.rollAcc * aaCfg.rollSpeed * math.pi / 180) % (2 * math.pi)
        end

        local basePos = root.Position
        if aaCfg.moveEnabled then
            if aaCfg.moveMode == "Camera" and Camera then
                basePos = basePos
                    + Camera.CFrame.LookVector * aaCfg.moveForward
                    + Camera.CFrame.RightVector * aaCfg.moveRight
                    + Vector3.new(0, aaCfg.moveUp, 0)
            else
                basePos = basePos + Vector3.new(aaCfg.moveRight, aaCfg.moveUp, aaCfg.moveForward)
            end
        end
        if aaCfg.spinOrbit then
            aaState.spinOrbitAng = aaState.spinOrbitAng + aaCfg.spinOrbitSpeed * dt
            basePos = basePos + Vector3.new(
                math.cos(aaState.spinOrbitAng) * aaCfg.spinOrbitRadius,
                0,
                math.sin(aaState.spinOrbitAng) * aaCfg.spinOrbitRadius
            )
        end
        if aaCfg.upDown then
            basePos = basePos + Vector3.new(0, math.sin(aaState.upDownT * aaCfg.upDownSpeed) * aaCfg.upDownAmp, 0)
        end

        local m = aaCfg.mode
        if m == "Spin" then
            aaState.angleAcc = aaState.angleAcc + spd * dt
            root.CFrame = CFrame.new(basePos) * CFrame.Angles(math.rad(extraPitch), math.rad(aaState.angleAcc + extraYaw), rollAngle)
        elseif m == "Jitter" then
            local p = aaCfg.pitchAngle
            local y = aaCfg.yawAngle
            root.CFrame = CFrame.new(basePos) * CFrame.Angles(
                math.rad((aaState.jitterState == 0 and p or -p) + extraPitch),
                math.rad((aaState.jitterState == 0 and y or -y) + extraYaw),
                rollAngle
            )
            aaState.jitterState = (aaState.jitterState + 1) % 2
        elseif m == "Static" then
            root.CFrame = CFrame.new(basePos) * CFrame.Angles(math.rad(aaCfg.staticPitch + extraPitch), math.rad(aaCfg.staticYaw + extraYaw), rollAngle)
        elseif m == "RandomSpin" then
            aaState.angleAcc = aaState.angleAcc + spd * dt * (math.random() * 2 - 1)
            root.CFrame = CFrame.new(basePos) * CFrame.Angles(math.rad(aaCfg.customPitch + extraPitch), math.rad(aaState.angleAcc + extraYaw), rollAngle)
        elseif m == "Custom" then
            root.CFrame = CFrame.new(basePos) * CFrame.Angles(math.rad(aaCfg.customPitch + extraPitch), math.rad(aaCfg.customYaw + extraYaw), rollAngle)
        elseif m == "Up" then
            root.CFrame = CFrame.new(basePos) * CFrame.Angles(math.rad(90 + extraPitch), math.rad(extraYaw), rollAngle)
        elseif m == "Down" then
            root.CFrame = CFrame.new(basePos) * CFrame.Angles(math.rad(-90 + extraPitch), math.rad(extraYaw), rollAngle)
        elseif m == "Side" then
            root.CFrame = CFrame.new(basePos) * CFrame.Angles(math.rad(extraPitch), math.rad(extraYaw), math.rad(90) + rollAngle)
        end
    end)
end

local function abInit(p)
    abState.data[p] = { lastPos = nil, tpTimes = {}, lastShot = 0 }
end

local function abClean(data)
    local now, clean = tick(), {}
    for _, t in ipairs(data.tpTimes) do
        if now - t < abCfg.flickerWindow then clean[#clean + 1] = t end
    end
    data.tpTimes = clean
end

local function abGetDodgeDir(threatPos)
    if not root then return Vector3.new(1, 0, 0) end
    local flat = Vector3.new(threatPos.X - root.Position.X, 0, threatPos.Z - root.Position.Z)
    local away = flat.Magnitude > 0.01 and -flat.Unit or Vector3.new(1, 0, 0)
    local right = Vector3.new(-away.Z, 0, away.X)
    local up = Vector3.new(0, 1, 0)
    local m = abCfg.dodgeMode
    if m == "Away" then return away
    elseif m == "Toward" then return -away
    elseif m == "Up" then return up
    elseif m == "Down" then return -up
    elseif m == "Diagonal" then
        local side = math.random(0, 1) == 0 and right or -right
        return (away + side).Unit
    elseif m == "Spiral" then
        local ang = tick() * 6
        return right * math.cos(ang) + away * math.sin(ang)
    elseif m == "Backstep" then return away + up * 0.4
    elseif m == "SideStep" then
        return math.random(0, 1) == 0 and right or -right
    end
    return math.random(0, 1) == 0 and right or -right
end

local function abDodge(p, reason, threatPos)
    if not root then return end
    local now = tick()
    if now - abState.lastDodge < abCfg.cooldown then return end
    abState.lastDodge = now

    local dir = abGetDodgeDir(threatPos or root.Position)
    local newPos = root.Position + dir * abCfg.dodgeDist
    if abCfg.verticalSpike then
        newPos = newPos + Vector3.new(0, math.random() * 20 + 5, 0)
    end
    pcall(function()
        root.CFrame = CFrame.new(newPos)
        root.AssemblyLinearVelocity = dir * 150
    end)

    if abCfg.notify then
        local dist = 0
        if p and p.Character then
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if hrp then dist = math.floor((hrp.Position - root.Position).Magnitude) end
        end
        Notify("anti-bait: " .. (p and p.Name or "?") .. " [" .. reason .. "] " .. dist .. " studs", 1.5)
    end
end

function startAntiBait()
    killConn("antiBait")
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP then abInit(p) end
    end
    Players.PlayerAdded:Connect(function(p) if p ~= LP then abInit(p) end end)
    Players.PlayerRemoving:Connect(function(p) abState.data[p] = nil end)

    conns.antiBait = RunService.Heartbeat:Connect(function(dt)
        if not abCfg.enabled then killConn("antiBait"); return end
        if not root then return end
        local myPos = root.Position
        local now = tick()

        for _, p in ipairs(Players:GetPlayers()) do
            if p == LP or not p.Character then continue end
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if not hrp then continue end
            local data = abState.data[p]
            if not data then abInit(p); data = abState.data[p] end

            local curPos = hrp.Position
            local vel = hrp.AssemblyLinearVelocity
            local dist = (curPos - myPos).Magnitude
            local effRadius = abCfg.infiniteRange and 5e12 or 500
            if dist > effRadius then data.lastPos = curPos; continue end

            if abCfg.velEnabled and vel.Magnitude > abCfg.velThreshold then
                local toMe = myPos - curPos
                if toMe.Magnitude > 0.01 and vel.Unit:Dot(toMe.Unit) > 0.6 then
                    abDodge(p, "vel-spike", curPos)
                end
            end

            if abCfg.gunDetect then
                local tool = p.Character:FindFirstChildOfClass("Tool")
                if tool then
                    local origin = getAimOrigin(p)
                    if origin then
                        local look = origin.CFrame.LookVector
                        local toMe = myPos - origin.Position
                        if toMe.Magnitude > 0.01 and look:Dot(toMe.Unit) > abCfg.gunDot then
                            abDodge(p, "gun", curPos)
                        end
                    end
                end
            end

            if abCfg.lookAtEnabled then
                local origin = getAimOrigin(p)
                if origin then
                    local look = origin.CFrame.LookVector
                    local toMe = myPos - origin.Position
                    if toMe.Magnitude > 0.01 and look:Dot(toMe.Unit) > abCfg.lookDotThresh then
                        abDodge(p, "look", curPos)
                    end
                end
            end

            if abCfg.aimPredict then
                local prediction = predictEnemyAim(p, abCfg.aimPredictLead, root)
                if prediction and prediction.futureDot > 0.85 and prediction.dist < effRadius then
                    abDodge(p, "predict", curPos)
                end
            end

            if abCfg.flickerEnabled and data.lastPos then
                if (curPos - data.lastPos).Magnitude > abCfg.flickerTpDist then
                    data.tpTimes[#data.tpTimes + 1] = now
                end
                abClean(data)
                if #data.tpTimes >= abCfg.flickerMinTps then
                    data.tpTimes = {}
                    abDodge(p, "flicker", curPos)
                end
            end

            data.lastPos = curPos
        end
    end)
end

function stopAntiBait()
    killConn("antiBait")
    abCfg.enabled = false
    abState.data = {}
end

function startFakePos()
    if fakeState.conn then fakeState.conn:Disconnect() end
    if fakeState.renderConn then fakeState.renderConn:Disconnect() end
    fakeState.realCF = nil
    fakeState.lastPacket = 0
    fakeState.spiralAng = 0
    fakeState.pulseT = 0

    fakeState.conn = RunService.Heartbeat:Connect(function(dt)
        if not fakeCfg.enabled or not root then return end
        fakeState.realCF = root.CFrame
        fakeState.pulseT = fakeState.pulseT + dt
        local now = tick()
        local spd = fakeCfg.randomizeEveryFrame and 9999 or math.max(fakeCfg.speed, 1)
        if now - fakeState.lastPacket < (1 / spd) then return end
        fakeState.lastPacket = now

        local r = math.max(fakeCfg.radius, 1)
        if fakeCfg.pulseRadius then
            r = fakeCfg.pulseRadiusMin + (math.sin(fakeState.pulseT * fakeCfg.pulseSpeed * 2 * math.pi) * 0.5 + 0.5) * (fakeCfg.pulseRadiusMax - fakeCfg.pulseRadiusMin)
        end

        local layers = fakeCfg.layeredFake and fakeCfg.layerCount or 1

        for layer = 1, layers do
            local ox, oy, oz = 0, 0, 0
            local lr = r * (layer / layers)
            if fakeCfg.chaosMode then lr = lr * (1 + (math.random() - 0.5) * fakeCfg.chaosMult) end

            if fakeCfg.pattern == "Scatter" then
                local a = (math.random() - 0.5) * math.rad(fakeCfg.scatterAngle)
                ox = math.cos(a) * (math.random() * lr)
                oy = fakeCfg.height + math.random() * fakeCfg.yDist
                oz = math.sin(a) * (math.random() * lr)
            elseif fakeCfg.pattern == "Linear" then
                ox = (math.random() - 0.5) * 2 * lr
                oy = fakeCfg.height + math.random() * fakeCfg.yDist
                oz = (math.random() - 0.5) * 2 * lr
            elseif fakeCfg.pattern == "Circle" then
                fakeState.spiralAng = (fakeState.spiralAng + math.rad(60)) % (2 * math.pi)
                ox = math.cos(fakeState.spiralAng) * lr
                oy = fakeCfg.height + math.random() * fakeCfg.yDist
                oz = math.sin(fakeState.spiralAng) * lr
            elseif fakeCfg.pattern == "Random" then
                ox = (math.random() * 2 - 1) * lr
                oy = fakeCfg.height + math.random() * fakeCfg.yDist
                oz = (math.random() * 2 - 1) * lr
            elseif fakeCfg.pattern == "Spiral" then
                fakeState.spiralAng = (fakeState.spiralAng + math.rad(60)) % (2 * math.pi)
                local sr = lr * (0.3 + 0.7 * ((fakeState.spiralAng / (2 * math.pi)) % 1))
                ox = math.cos(fakeState.spiralAng) * sr
                oy = fakeCfg.height + math.random() * fakeCfg.yDist
                oz = math.sin(fakeState.spiralAng) * sr
            end

            if fakeCfg.distribution == "Uniform" then
                local mag = math.sqrt(ox * ox + oz * oz)
                if mag > 0 then local s = lr / mag; ox = ox * s; oz = oz * s end
            elseif fakeCfg.distribution == "Weighted" then
                ox = ox * 0.6; oz = oz * 0.6
            elseif fakeCfg.distribution == "Radial" then
                local theta = math.atan2(oz, ox)
                ox = math.cos(theta) * lr
                oz = math.sin(theta) * lr
            elseif fakeCfg.distribution == "Clustered" then
                local ca = math.floor(math.random() * 8) * (math.pi * 0.25)
                ox = math.cos(ca) * lr
                oz = math.sin(ca) * lr
            end

            if fakeCfg.axisLock == "X" then oz = 0; oy = fakeCfg.height
            elseif fakeCfg.axisLock == "Y" then ox = 0; oz = 0
            elseif fakeCfg.axisLock == "Z" then ox = 0; oy = fakeCfg.height
            elseif fakeCfg.axisLock == "XZ" then oy = fakeCfg.height end

            local fcf = fakeState.realCF + Vector3.new(ox, oy, oz)
            pcall(function()
                root.AssemblyLinearVelocity = Vector3.zero
                root.CFrame = fcf
                root.AssemblyLinearVelocity = Vector3.zero
            end)
        end
    end)

    fakeState.renderConn = RunService.RenderStepped:Connect(function()
        if fakeCfg.enabled and root and fakeState.realCF then
            root.CFrame = fakeState.realCF
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
        end
    end)
end

function stopFakePos()
    if fakeState.conn then fakeState.conn:Disconnect(); fakeState.conn = nil end
    if fakeState.renderConn then fakeState.renderConn:Disconnect(); fakeState.renderConn = nil end
    fakeState.realCF = nil
    fakeCfg.enabled = false
end

local function fmt(n)
    local a = math.abs(n)
    if a >= 1e9 then
        return string.format("%.2fB", n / 1e9)
    elseif a >= 1e6 then
        return string.format("%.2fM", n / 1e6)
    elseif a >= 1e4 then
        return string.format("%.1fK", n / 1e3)
    elseif a >= 100 then
        return string.format("%.1f", n)
    else
        return string.format("%.2f", n)
    end
end
